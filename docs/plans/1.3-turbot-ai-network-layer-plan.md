## 4. 核心接口设计

### 4.1 HTTP 客户端

> **实现说明**: 实际实现做了以下改进：
>
> - 使用 `HttpMethod` 枚举替代字符串，类型更安全
> - 添加 Builder 模式和静态工厂方法，提高易用性
> - 流式响应使用回调模式，比协程生成器更简单实用
> - 异步使用 `std::future`，减少 Asio 依赖
> - 添加 SSL 验证开关、User-Agent 配置等增强功能

**文件**: `libs/network/include/turbot/network/http_client.hpp`

```cpp
namespace turbot::network {

// HTTP 方法枚举（类型安全）
enum class HttpMethod {
    GET, POST, PUT, PATCH, DELETE_, HEAD, OPTIONS
};

// HTTP 请求
struct HttpRequest {
    HttpMethod method = HttpMethod::GET;
    std::string url;
    HttpHeaders headers;
    std::string body;
    int timeout_seconds = 30;
    bool follow_redirects = true;
    int max_redirects = 5;

    // 静态工厂方法
    static HttpRequest get(std::string_view url);
    static HttpRequest post(std::string_view url, std::string_view body = "");
    static HttpRequest put(std::string_view url, std::string_view body = "");
    static HttpRequest patch(std::string_view url, std::string_view body = "");
    static HttpRequest del(std::string_view url);

    // Builder 模式
    HttpRequest& with_header(std::string_view name, std::string_view value);
    HttpRequest& with_timeout(int seconds);
    HttpRequest& with_json_body(std::string_view json);
};

// HTTP 响应
struct HttpResponse {
    int status_code = 0;
    std::string status_message;
    std::string body;
    HttpHeaders headers;
    std::string content_type;
    long response_time_ms = 0;  // 响应时间（毫秒）

    // 状态检查方法
    bool is_success() const noexcept;      // 2xx
    bool is_client_error() const noexcept; // 4xx
    bool is_server_error() const noexcept; // 5xx

    // 获取响应头（大小写不敏感）
    std::optional<std::string> get_header(std::string_view name) const;
};

// 流式响应回调
tusing StreamCallback = std::function<bool(std::string_view chunk)>;

class HttpClient {
public:
    HttpClient();
    ~HttpClient();

    // 高级 API
    HttpResponse get(std::string_view url, const HttpHeaders& headers = {});
    HttpResponse post(std::string_view url, std::string_view body, const HttpHeaders& headers = {});
    HttpResponse post_json(std::string_view url, std::string_view json, const HttpHeaders& headers = {});
    HttpResponse put(std::string_view url, std::string_view body, const HttpHeaders& headers = {});
    HttpResponse patch(std::string_view url, std::string_view body, const HttpHeaders& headers = {});
    HttpResponse del(std::string_view url, const HttpHeaders& headers = {});

    // 低级 API
    HttpResponse request(const HttpRequest& req);
    HttpResponse request_stream(const HttpRequest& req, StreamCallback callback);

    // 异步 API（使用 std::future）
    std::future<HttpResponse> get_async(std::string_view url, const HttpHeaders& headers = {});
    std::future<HttpResponse> post_async(std::string_view url, std::string_view body, const HttpHeaders& headers = {});

    // 配置
    void set_timeout(int seconds);
    void set_default_header(std::string_view name, std::string_view value);
    void set_proxy(std::string_view proxy);
    void set_ssl_verify(bool verify);
    void set_user_agent(std::string_view user_agent);
};

} // namespace turbot::network
```

enum class WebSocketState {
    Connecting,
    Open,
    Closing,
    Closed
};

struct WebSocketMessage {
    enum class Type { Text, Binary, Close, Ping, Pong } type;
    std::string data;
};

class WebSocketClient {
public:
    using MessageHandler = std::function<void(const WebSocketMessage&)>;
    using CloseHandler = std::function<void(int code, const std::string& reason)>;

    WebSocketClient();
    ~WebSocketClient();

    // 连接WebSocket
    Task<bool> connect(const std::string& url,
                       const std::vector<std::pair<std::string, std::string>>& headers = {});

    // 断开连接
    Task<void> close(int code = 1000, const std::string& reason = "");

    // 发送消息
    Task<void> send(const std::string& message);
    Task<void> send_binary(const std::vector<uint8_t>& data);

    // 设置消息处理器
    void on_message(MessageHandler handler) { message_handler_ = handler; }

    // 设置关闭处理器
    void on_close(CloseHandler handler) { close_handler_ = handler; }

    // 获取状态
    WebSocketState state() const { return state_; }

    // 是否已连接
    bool is_connected() const { return state_ == WebSocketState::Open; }

private:
    WebSocketState state_ = WebSocketState::Closed;
    MessageHandler message_handler_;
    CloseHandler close_handler_;
    std::unique_ptr<asio::io_context> io_context_;
    std::unique_ptr<websocket::stream<tcp::socket>> ws_;
};

} // namespace turbot::network::websocket

## 5. 实现要点

### 5.1 HTTP 客户端实现

```cpp
Task<HttpResponse> HttpClient::request(const HttpRequest& req) {
    HttpResponse response;

    // 创建libcurl句柄
    CURL* curl = curl_easy_init();
    if (!curl) {
        throw std::runtime_error("Failed to initialize curl");
    }

    // 设置URL
    curl_easy_setopt(curl, CURLOPT_URL, req.url.c_str());

    // 设置方法
    curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, req.method.c_str());

    // 设置超时
    curl_easy_setopt(curl, CURLOPT_TIMEOUT, req.timeout);

    // 设置跟随重定向
    if (req.follow_redirects) {
        curl_easy_setopt(curl, CURLOPT_FOLLOWLOCATION, 1L);
        curl_easy_setopt(curl, CURLOPT_MAXREDIRS, req.max_redirects);
    }

    // 设置请求头
    struct curl_slist* headers = nullptr;
    for (const auto& [key, value] : req.headers) {
        std::string header = key + ": " + value;
        headers = curl_slist_append(headers, header.c_str());
    }

    // 添加默认头
    for (const auto& [key, value] : default_headers_) {
        std::string header = key + ": " + value;
        headers = curl_slist_append(headers, header.c_str());
    }

    curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers);

    // 设置请求体
    if (!req.body.empty()) {
        curl_easy_setopt(curl, CURLOPT_POSTFIELDS, req.body.c_str());
        curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE, req.body.size());
    }

    // 设置响应处理
    std::string response_body;
    curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, write_callback);
    curl_easy_setopt(curl, CURLOPT_WRITEDATA, &response_body);

    // 设置响应头处理
    std::string response_headers;
    curl_easy_setopt(curl, CURLOPT_HEADERFUNCTION, header_callback);
    curl_easy_setopt(curl, CURLOPT_HEADERDATA, &response_headers);

    // 执行请求
    CURLcode res = curl_easy_perform(curl);

    if (res != CURLE_OK) {
        curl_slist_free_all(headers);
        curl_easy_cleanup(curl);
        throw std::runtime_error("curl_easy_perform() failed: " + std::string(curl_easy_strerror(res)));
    }

    // 获取响应码
    long status_code;
    curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, &status_code);
    response.status_code = static_cast<int>(status_code);

    // 解析响应头
    response.headers = parse_headers(response_headers);

    // 设置响应体
    response.body = response_body;

    // 清理
    curl_slist_free_all(headers);
    curl_easy_cleanup(curl);

    co_return response;
}

size_t write_callback(void* contents, size_t size, size_t nmemb, void* userp) {
    size_t total_size = size * nmemb;
    std::string* response = static_cast<std::string*>(userp);
    response->append(static_cast<char*>(contents), total_size);
    return total_size;
}

size_t header_callback(char* buffer, size_t size, size_t nitems, void* userdata) {
    size_t total_size = size * nitems;
    std::string* headers = static_cast<std::string*>(userdata);
    headers->append(buffer, total_size);
    return total_size;
}
```

### 5.2 流式响应处理

```cpp
Task<StreamResponse> HttpClient::request_stream(const HttpRequest& req) {
    StreamResponse response;

    CURL* curl = curl_easy_init();
    if (!curl) {
        throw std::runtime_error("Failed to initialize curl");
    }

    curl_easy_setopt(curl, CURLOPT_URL, req.url.c_str());
    curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, req.method.c_str());
    curl_easy_setopt(curl, CURLOPT_TIMEOUT, req.timeout);

    // 设置流式响应
    curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, nullptr);
    curl_easy_setopt(curl, CURLOPT_WRITEDATA, nullptr);

    // 创建异步生成器
    auto generator = co_await create_stream_generator(curl, req);

    // 获取响应头
    long status_code;
    curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, &status_code);
    response.status_code = static_cast<int>(status_code);

    response.body_stream = std::move(generator);

    co_return response;
}
```

## 7. 使用示例

### 7.1 HTTP 请求

```cpp
#include "turbot/network/http/http_client.hpp"

using namespace turbot::network::http;

// 创建客户端
HttpClient client;

// GET请求
auto response = co_await client.get("https://api.openai.com/v1/models", {
    {"Authorization", "Bearer sk-xxx"}
});

if (response.is_success()) {
    auto json = nlohmann::json::parse(response.body);
    TURBOT_LOG_INFO("Models: {}", json.dump());
}

// POST请求
nlohmann::json body = {
    {"model", "gpt-4"},
    {"messages", nlohmann::json::array({
        {{"role", "user"}, {"content", "Hello"}}
    })}
};

auto response = co_await client.post("https://api.openai.com/v1/chat/completions", body, {
    {"Authorization", "Bearer sk-xxx"},
    {"Content-Type", "application/json"}
});
```
