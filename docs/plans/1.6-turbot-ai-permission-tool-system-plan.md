## 2. 权限系统

### 2.1 权限接口

```cpp
namespace turbot::core::permission {

enum class PermissionAction {
    Allow,
    Deny,
    Ask
};

struct PermissionRule {
    std::string permission;  // 权限类型
    std::string pattern;     // 匹配模式
    PermissionAction action;
};

using Ruleset = std::vector<PermissionRule>;

struct PermissionRequest {
    std::string permission;
    std::vector<std::string> patterns;
    std::optional<std::string> tool;
    nlohmann::json metadata;
};

struct PermissionReply {
    enum class Type { Once, Always, Reject } type;
    std::optional<std::string> message;
};

class PermissionSystem {
public:
    // 评估权限
    PermissionAction evaluate(const std::string& permission,
                             const std::string& pattern,
                             const Ruleset& ruleset) const;

    // 合并规则集
    static Ruleset merge(const std::vector<Ruleset>& rulesets);

    // 通配符匹配
    static bool wildcard_match(const std::string& pattern,
                               const std::string& text);

    // 请求权限
    Task<PermissionReply> ask(const PermissionRequest& request);

    // 回复权限请求
    Task<void> reply(const std::string& request_id,
                     const PermissionReply& reply);
};

} // namespace turbot::core::permission
```

### 2.2 权限评估逻辑

```cpp
PermissionAction PermissionSystem::evaluate(
    const std::string& permission,
    const std::string& pattern,
    const Ruleset& ruleset) const {

    PermissionAction result = PermissionAction::Deny;

    for (const auto& rule : ruleset) {
        if (wildcard_match(rule.pattern, pattern) &&
            wildcard_match(rule.permission, permission)) {
            result = rule.action;
        }
    }

    return result;
}

bool PermissionSystem::wildcard_match(const std::string& pattern,
                                       const std::string& text) {
    // 简单通配符匹配 (*匹配任意字符序列)
    size_t i = 0, j = 0;
    size_t star_pos = std::string::npos;
    size_t match_pos = 0;

    while (i < text.size()) {
        if (j < pattern.size() && (pattern[j] == text[i] || pattern[j] == '?')) {
            i++;
            j++;
        } else if (j < pattern.size() && pattern[j] == '*') {
            star_pos = j;
            match_pos = i;
            j++;
        } else if (star_pos != std::string::npos) {
            j = star_pos + 1;
            match_pos++;
            i = match_pos;
        } else {
            return false;
        }
    }

    while (j < pattern.size() && pattern[j] == '*') {
        j++;
    }

    return j == pattern.size();
}
```

### 3.1 工具接口

```cpp
namespace turbot::core::tool {

struct ToolContext {
    std::string session_id;
    std::string message_id;
    std::string agent;
    std::optional<std::string> call_id;
    std::function<void(const nlohmann::json&)> metadata;
    std::function<Task<void>(const PermissionRequest&)> ask;
    std::vector<message::Message> messages;
    std::shared_ptr<std::atomic<bool>> abort_flag;
};

struct ToolResult {
    std::string title;
    std::string output;
    nlohmann::json metadata;
};

class Tool {
public:
    virtual ~Tool() = default;

    virtual std::string name() const = 0;
    virtual std::string description() const = 0;
    virtual nlohmann::json input_schema() const = 0;

    virtual Task<ToolResult> execute(const nlohmann::json& input,
                                     ToolContext& ctx) = 0;

    virtual bool validate_input(const nlohmann::json& input) const;
    virtual std::string format_validation_error(const std::string& error) const;
};

} // namespace turbot::core::tool
```

### 3.3 基础工具示例

```cpp
// 文件读取工具
class ReadFileTool : public Tool {
public:
    std::string name() const override { return "read_file"; }

    std::string description() const override {
        return "Read the contents of a file";
    }

    nlohmann::json input_schema() const override {
        return R"({
            "type": "object",
            "properties": {
                "path": {"type": "string"}
            },
            "required": ["path"]
        })"_json;
    }

    Task<ToolResult> execute(const nlohmann::json& input,
                             ToolContext& ctx) override {
        std::string path = input["path"];

        // 权限检查
        auto result = co_await ctx.ask({
            .permission = "read",
            .patterns = {path},
            .tool = name()
        });

        if (result.type != PermissionReply::Type::Always) {
            throw std::runtime_error("Permission denied");
        }

        // 读取文件
        std::ifstream file(path);
        if (!file) {
            throw std::runtime_error("Cannot open file: " + path);
        }

        std::string content((std::istreambuf_iterator<char>(file)),
                            std::istreambuf_iterator<char>());

        co_return ToolResult{
            .title = "Read file: " + path,
            .output = content
        };
    }
};
```

## 4. 使用示例

```cpp
// 注册工具
ToolRegistry::instance().register_tool(std::make_unique<ReadFileTool>());
ToolRegistry::instance().register_tool(std::make_unique<WriteFileTool>());

// 创建权限规则
Ruleset ruleset = {
    {"read", "*", PermissionAction::Allow},
    {"write", "/allowed/*", PermissionAction::Allow},
    {"write", "*", PermissionAction::Ask}
};

// 评估权限
auto action = PermissionSystem::evaluate("read", "/path/to/file.txt", ruleset);
```
