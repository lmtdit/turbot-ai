# Turbot-AI 核心数据模型实现计划

## 1. 概述

本计划是 Turbot-AI 项目的第四阶段实现，旨在建立核心数据模型。核心数据模型包括Message和Part系统，是整个项目的数据基础，用于表示AI对话中的消息和消息的各个组成部分。

### 1.1 目标

- 实现Message数据模型
- 实现12种Part类型
- 实现消息持久化
- 实现Token统计
- 实现消息查询和检索

### 1.2 依赖关系

- **前置依赖**: 1.1 - 基础架构计划, 1.2 - 存储层计划
- **后置依赖**: Provider系统、会话系统、Agent系统

### 1.3 交付物

- Message数据模型
- Part系统实现
- 消息DAO层
- Token统计功能
- 单元测试和集成测试

### 3.1 Token使用统计

**文件**: `libs/core/message/token_usage.hpp`

```cpp
namespace turbot::core::message {

struct TokenUsage {
    int64_t input = 0;
    int64_t output = 0;
    int64_t reasoning = 0;
    struct {
        int64_t read = 0;
        int64_t write = 0;
    } cache;

    int64_t total() const {
        return input + output + reasoning + cache.read + cache.write;
    }

    double cost(const nlohmann::json& pricing) const {
        double total_cost = 0.0;
        total_cost += input * pricing.value("input", 0.0);
        total_cost += output * pricing.value("output", 0.0);
        total_cost += reasoning * pricing.value("reasoning", 0.0);
        total_cost += cache.read * pricing.value("cache_read", 0.0);
        total_cost += cache.write * pricing.value("cache_write", 0.0);
        return total_cost;
    }
};

} // namespace turbot::core::message
```

### 3.3 Message模型

**文件**: `libs/core/message/message.hpp`

```cpp
namespace turbot::core::message {

enum class Role {
    User,
    Assistant,
    System
};

struct MessageInfo {
    std::string id;
    std::string session_id;
    Role role;
    int64_t time_created;
    int64_t time_updated;
    std::optional<std::string> parent_id;
    std::string agent;
    std::string model_id;
    std::string provider_id;
    std::optional<std::string> system;
    std::optional<nlohmann::json> tools;
    std::optional<std::string> variant;
    std::optional<nlohmann::json> error;
    std::optional<std::string> finish;
    double cost = 0.0;
    TokenUsage tokens;
    std::optional<bool> summary;
    std::optional<nlohmann::json> structured;
};

class Message {
public:
    // 创建消息
    static Task<Message> create(const std::string& session_id, Role role,
                                const std::string& agent,
                                const std::string& model_id,
                                const std::string& provider_id);

    // 获取消息
    static Task<std::optional<Message>> get(const std::string& id);

    // 获取会话的消息流
    static Task<std::vector<Message>> stream(const std::string& session_id,
                                             int limit = 50, int offset = 0);

    // 查询消息
    static Task<std::vector<Message>> query(const std::string& session_id,
                                            const nlohmann::json& filter);

    // 保存消息
    Task<void> save();

    // 更新消息
    Task<void> update();

    // 删除消息
    Task<void> remove();

    // 添加Part
    void add_part(const Part& part);

    // 获取指定类型的Parts
    std::vector<Part> get_parts(PartType type) const;

    // 获取文本内容
    std::string get_text() const;

    // 获取工具调用
    std::vector<nlohmann::json> get_tool_calls() const;

    // 判断是否包含工具调用
    bool has_tool_calls() const;

    // 获取完整文本（包括所有TextPart）
    std::string get_full_text() const;

    // 设置错误
    void set_error(const nlohmann::json& error);

    // 设置完成状态
    void set_finish(const std::string& finish_reason);

    // 更新Token使用
    void update_tokens(const TokenUsage& tokens, const nlohmann::json& pricing);

    // Getter
    const std::string& id() const { return info_.id; }
    const std::string& session_id() const { return info_.session_id; }
    Role role() const { return info_.role; }
    const MessageInfo& info() const { return info_; }
    const std::vector<Part>& parts() const { return parts_; }

private:
    MessageInfo info_;
    std::vector<Part> parts_;
    std::shared_ptr<storage::Database> db_;
};

} // namespace turbot::core::message
```

Part Part::create_text(const std::string& content) {
    Part part;
    part.id = generate_uuid();
    part.type = PartType::Text;
    part.data = {{"content", content}};
    part.time_created = part.time_updated = current_timestamp();
    return part;
}

Part Part::create_tool(const std::string& tool, const nlohmann::json& params,
                       const nlohmann::json& result) {
    Part part;
    part.id = generate_uuid();
    part.type = PartType::Tool;
    part.data = {
        {"tool", tool},
        {"params", params},
        {"result", result}
    };
    part.time_created = part.time_updated = current_timestamp();
    return part;
}

std::string Part::get_text() const {
    if (type == PartType::Text) {
        return data.value("content", "");
    }
    return "";
}

nlohmann::json Part::get_tool() const {
    if (type == PartType::Tool) {
        return data;
    }
    return nlohmann::json{};
}

### 4.2 Message实现

```cpp
Task<Message> Message::create(const std::string& session_id, Role role,
                              const std::string& agent,
                              const std::string& model_id,
                              const std::string& provider_id) {
    Message message;
    message.info_.id = generate_uuid();
    message.info_.session_id = session_id;
    message.info_.role = role;
    message.info_.agent = agent;
    message.info_.model_id = model_id;
    message.info_.provider_id = provider_id;
    message.info_.time_created = message.info_.time_updated = current_timestamp();
    message.db_ = get_default_database();

    co_await message.save();
    co_return message;
}

Task<void> Message::save() {
    dao::MessageDAO dao(db_);
    co_await dao.create(info_);

    for (auto& part : parts_) {
        part.message_id = info_.id;
        part.session_id = info_.session_id;
        co_await dao.create_part(part);
    }
}

std::string Message::get_text() const {
    std::string text;
    for (const auto& part : parts_) {
        if (part.is_text()) {
            text += part.get_text();
        }
    }
    return text;
}

std::vector<nlohmann::json> Message::get_tool_calls() const {
    std::vector<nlohmann::json> calls;
    for (const auto& part : parts_) {
        if (part.is_tool()) {
            calls.push_back(part.get_tool());
        }
    }
    return calls;
}
```

## 7. 使用示例

```cpp
#include "turbot/core/message/message.hpp"

using namespace turbot::core::message;

// 创建用户消息
auto user_message = co_await Message::create(
    "session123", Role::User, "build", "gpt-4", "openai"
);

// 添加文本Part
user_message.add_part(Part::create_text("Write a function to add two numbers"));

// 保存
co_await user_message.save();

// 创建助手消息
auto assistant_message = co_await Message::create(
    "session123", Role::Assistant, "build", "gpt-4", "openai"
);

// 添加推理Part
assistant_message.add_part(Part::create_reasoning(
    "I need to create a simple addition function in C++"
));

// 添加文本Part
assistant_message.add_part(Part::create_text(
    "```cpp\nint add(int a, int b) {\n    return a + b;\n}\n```"
));

// 保存
co_await assistant_message.save();

// 查询会话的所有消息
auto messages = co_await Message::stream("session123");

for (const auto& msg : messages) {
    TURBOT_LOG_INFO("Message from {}: {}",
        msg.role() == Role::User ? "user" : "assistant",
        msg.get_text());
}
```
