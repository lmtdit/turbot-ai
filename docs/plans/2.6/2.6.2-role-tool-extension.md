### Step 2: 扩展 `role_to_string` / `role_from_string`

**文件**: `libs/core/src/message/part.cpp`

```cpp
std::string_view role_to_string(Role role) noexcept {
    switch (role) {
        case Role::User:      return "user";
        case Role::Assistant: return "assistant";
        case Role::System:    return "system";
        case Role::Tool:      return "tool";      // 新增
    }
    return "unknown";
}

Role role_from_string(std::string_view str) {
    if (str == "user")      return Role::User;
    if (str == "assistant") return Role::Assistant;
    if (str == "system")    return Role::System;
    if (str == "tool")      return Role::Tool;    // 新增
    throw std::invalid_argument(std::string("Invalid Role: ") + std::string(str));
}
```
