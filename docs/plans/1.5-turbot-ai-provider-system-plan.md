## 3. 核心接口

### 3.1 Provider 接口

```cpp
namespace turbot::core::provider {

struct ModelConfig {
    std::string model_id;
    std::string provider_id;
};

struct ModelCapabilities {
    bool temperature = false;
    bool reasoning = false;
    bool tool_call = false;
    bool streaming = false;
};

struct ModelInfo {
    std::string id;
    std::string provider_id;
    std::string name;
    ModelCapabilities capabilities;
    nlohmann::json pricing;
    nlohmann::json limits;
};

struct ChatMessage {
    enum class Role { System, User, Assistant, Tool };
    Role role;
    std::string content;
    std::optional<std::string> tool_call_id;
    std::optional<nlohmann::json> tool_calls;
};

struct ChatStreamEvent {
    enum class Type { TextDelta, ToolCall, Finish, Error };
    Type type;
    std::string content;
    nlohmann::json data;
};

class Provider {
public:
    virtual ~Provider() = default;

    virtual std::string name() const = 0;

    // 获取模型信息
    virtual Task<ModelInfo> get_model(const std::string& model_id) = 0;

    // 列出所有模型
    virtual Task<std::vector<ModelInfo>> list_models() = 0;

    // 聊天请求（流式）
    virtual AsyncGenerator<ChatStreamEvent> chat_stream(
        const std::vector<ChatMessage>& messages,
        const ModelConfig& config,
        const nlohmann::json& options = {}
    ) = 0;

    // 聊天请求（非流式）
    virtual Task<ChatMessage> chat(
        const std::vector<ChatMessage>& messages,
        const ModelConfig& config,
        const nlohmann::json& options = {}
    ) = 0;

    // 验证认证
    virtual Task<bool> authenticate() = 0;
};

} // namespace turbot::core::provider
```

### 3.2 ProviderManager

```cpp
class ProviderManager {
```

## 4. OpenAI 实现

```cpp
namespace turbot::core::provider::openai {

class OpenAIProvider : public Provider {
public:
    explicit OpenAIProvider(const std::string& api_key);

    std::string name() const override { return "openai"; }

    Task<ModelInfo> get_model(const std::string& model_id) override;

    Task<std::vector<ModelInfo>> list_models() override;

    AsyncGenerator<ChatStreamEvent> chat_stream(
        const std::vector<ChatMessage>& messages,
        const ModelConfig& config,
        const nlohmann::json& options = {}
    ) override;

    Task<ChatMessage> chat(
        const std::vector<ChatMessage>& messages,
        const ModelConfig& config,
        const nlohmann::json& options = {}
    ) override;

    Task<bool> authenticate() override;

private:
    std::string api_key_;
    std::shared_ptr<network::http::HttpClient> http_client_;
};

} // namespace turbot::core::provider::openai
```

## 5. 使用示例

```cpp
#include <turbot/core/provider/provider.hpp>
#include <turbot/core/provider/openai_provider.hpp>
#include <turbot/core/provider/bailian_provider.hpp>
#include <turbot/core/provider/provider_manager.hpp>

using namespace turbot::core::provider;

// 创建提供商
auto openai = std::make_shared<OpenAIProvider>("sk-xxx");
auto bailian = std::make_shared<BailianProvider>("your-bailian-key");

// 注册到管理器
ProviderManager::instance().register_provider(openai);
ProviderManager::instance().register_provider(bailian);
ProviderManager::instance().set_default_provider("openai");

// 聊天
std::vector<ChatMessage> messages = {
    ChatMessage::user("Hello, how are you?")
};

ChatOptions options;
options.temperature = 0.7;
options.max_tokens = 1000;

// 非流式响应
auto response = openai->chat(messages, "gpt-4o", options);
if (!response.is_error()) {
    std::cout << response.get_text() << std::endl;
}

// 流式响应
auto stream_response = openai->chat_stream(messages, "gpt-4o", options,
    [](const ChatStreamEvent& event) -> bool {
        if (event.type == StreamEventType::TextDelta) {
            std::cout << event.content << std::flush;
        }
        return true;  // 继续流
    }
);

// 使用 ProviderManager 查找模型
auto model = ProviderManager::instance().find_model("qwen-max");
if (model) {
    std::cout << "Found model: " << model->name << std::endl;
}
```
