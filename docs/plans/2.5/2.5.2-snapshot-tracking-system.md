## 目标

实现快照追踪系统，用于记录 Agent 执行过程中的文件变更，支持变更追踪、Patch 生成和回滚。

## 功能说明

快照追踪用于记录 Agent 执行过程中的文件变更，支持：

1. **变更追踪**: 记录文件创建/修改/删除
2. **Patch 生成**: 生成文件变更的差异
3. **回滚支持**: 支持撤销变更

## 接口设计

```cpp
// libs/core/snapshot/snapshot.hpp

#pragma once

#include <turbot/core/common/export.hpp>
#include <string>
#include <vector>
#include <optional>
#include <filesystem>

namespace turbot::core::snapshot {

/// 文件变更类型
enum class FileChangeType {
    Created,
    Modified,
    Deleted
};

/// 文件变更记录
struct TURBOT_CORE_API FileChange {
    std::filesystem::path path;
    FileChangeType type;
    std::string old_content;    // 对于 Modified/Deleted
    std::string new_content;    // 对于 Created/Modified
    std::string hash;           // 内容哈希
};

/// Patch 结果
struct TURBOT_CORE_API PatchResult {
    std::string hash;
    std::vector<FileChange> files;
};

/// 快照管理器
class TURBOT_CORE_API SnapshotManager {
public:
    static SnapshotManager& instance();

    /// 开始追踪
    std::string start_tracking();

    /// 结束追踪并生成 patch
    PatchResult stop_tracking(const std::string& snapshot_id);

    /// 应用 patch
    bool apply_patch(const PatchResult& patch);

    /// 回滚 patch
    bool rollback_patch(const PatchResult& patch);

private:
    SnapshotManager() = default;

    struct SnapshotData {
        std::unordered_map<std::string, std::string> file_hashes;
        std::chrono::system_clock::time_point start_time;
    };

    std::unordered_map<std::string, SnapshotData> snapshots_;
};

} // namespace turbot::core::snapshot
```

**snapshot.cpp**:

```cpp
#include <turbot/core/snapshot/snapshot.hpp>
#include <turbot/core/common/logger.hpp>
#include <fstream>
#include <sstream>
#include <openssl/sha.h>

namespace turbot::core::snapshot {

std::string SnapshotManager::start_tracking() {
    std::string snapshot_id = generate_uuid();

    SnapshotData data;
    data.start_time = std::chrono::system_clock::now();

    // 扫描当前目录文件哈希
    for (const auto& entry : std::filesystem::recursive_directory_iterator(".")) {
        if (entry.is_regular_file()) {
            data.file_hashes[entry.path().string()] = compute_file_hash(entry.path());
        }
    }

    snapshots_[snapshot_id] = std::move(data);
    return snapshot_id;
}

PatchResult SnapshotManager::stop_tracking(const std::string& snapshot_id) {
    auto it = snapshots_.find(snapshot_id);
    if (it == snapshots_.end()) {
        return {};
    }

    PatchResult result;
    result.hash = snapshot_id;

    // 对比当前文件状态
    for (const auto& entry : std::filesystem::recursive_directory_iterator(".")) {
        if (!entry.is_regular_file()) continue;

        std::string path = entry.path().string();
        std::string current_hash = compute_file_hash(entry.path());

        auto old_it = it->second.file_hashes.find(path);
        if (old_it == it->second.file_hashes.end()) {
            // 新文件
            result.files.push_back({
                .path = entry.path(),
                .type = FileChangeType::Created,
                .new_content = read_file_content(entry.path()),
                .hash = current_hash
            });
        } else if (old_it->second != current_hash) {
            // 修改的文件
            result.files.push_back({
                .path = entry.path(),
                .type = FileChangeType::Modified,
                .old_content = "", // 从备份获取
                .new_content = read_file_content(entry.path()),
                .hash = current_hash
            });
        }
    }

    // 检查删除的文件
    for (const auto& [path, hash] : it->second.file_hashes) {
        if (!std::filesystem::exists(path)) {
            result.files.push_back({
                .path = path,
                .type = FileChangeType::Deleted,
                .hash = hash
            });
        }
    }

    snapshots_.erase(it);
    return result;
}

} // namespace turbot::core::snapshot
```

### Step 4: 单元测试

```cpp
// tests/unit/snapshot/snapshot_test.cpp

TEST_CASE("SnapshotManager: Track file creation", "[snapshot]") {
    auto& manager = SnapshotManager::instance();

    std::string id = manager.start_tracking();

    // 创建新文件
    std::ofstream("test_new.txt") << "new content";

    auto patch = manager.stop_tracking(id);

    REQUIRE(patch.files.size() == 1);
    CHECK(patch.files[0].type == FileChangeType::Created);
    CHECK(patch.files[0].new_content == "new content");

    std::filesystem::remove("test_new.txt");
}

TEST_CASE("SnapshotManager: Track file modification", "[snapshot]") {
    // 创建初始文件
    std::ofstream("test_modify.txt") << "original";

    auto& manager = SnapshotManager::instance();
    std::string id = manager.start_tracking();

    // 修改文件
    std::ofstream("test_modify.txt") << "modified";

    auto patch = manager.stop_tracking(id);

    REQUIRE(patch.files.size() == 1);
    CHECK(patch.files[0].type == FileChangeType::Modified);

    std::filesystem::remove("test_modify.txt");
}

TEST_CASE("SnapshotManager: Track file deletion", "[snapshot]") {
    // 创建初始文件
    std::ofstream("test_delete.txt") << "to be deleted";

    auto& manager = SnapshotManager::instance();
    std::string id = manager.start_tracking();

    // 删除文件
    std::filesystem::remove("test_delete.txt");

    auto patch = manager.stop_tracking(id);

    REQUIRE(patch.files.size() == 1);
    CHECK(patch.files[0].type == FileChangeType::Deleted);
}
```
