## 1. OpenCode 核心流程分析

### 1.1 主循环 (prompt.ts:loop)

```typescript
// OpenCode 核心循环
export const loop = fn(LoopInput, async (input) => {
  const abort = start(sessionID)

  let step = 0
  while (true) {
    // 1. 获取消息历史
    let msgs = await MessageV2.filterCompacted(MessageV2.stream(sessionID))

    // 2. 查找最后一条用户消息和助手消息
    let lastUser, lastAssistant, lastFinished

    // 3. 检查是否应该退出循环
    if (
      lastAssistant?.finish &&
      !['tool-calls', 'unknown'].includes(lastAssistant.finish)
    ) {
      break
    }

    // 4. 获取模型配置
    const model = await Provider.getModel(
      lastUser.model.providerID,
      lastUser.model.modelID
    )

    // 5. 处理 pending subtask
    if (task?.type === 'subtask') {
      // 执行子任务
    }

    // 6. 检查是否需要压缩
    if (await SessionCompaction.isOverflow({ tokens, model })) {
      await SessionCompaction.create({ sessionID, agent, model, auto: true })
      continue
    }

    // 7. 创建 SessionProcessor 处理流式响应
    const processor = SessionProcessor.create({
      assistantMessage,
      sessionID,
      model,
      abort,
    })

    // 8. 处理流式响应
    const result = await processor.process(streamInput)

    // 9. 根据结果决定下一步
    if (result === 'compact') {
      /* 压缩 */
    }
    if (result === 'stop') break
    if (result === 'continue') continue
  }
})
```

### 1.3 LLM 调用 (llm.ts)

```typescript
export namespace LLM {
  export async function stream(input: StreamInput) {
    // 1. 获取 Provider 和认证
    const [language, cfg, provider, auth] = await Promise.all([...])

    // 2. 构建系统提示词
    const system = [
      input.agent.prompt ?? SystemPrompt.provider(input.model),
      ...input.system,
      input.user.system,
    ].filter(Boolean)

    // 3. 构建工具定义
    const tools = await resolveTools(input)

    // 4. 调用 AI SDK
    return streamText({
      model: language,
      messages: [...system.map(s => ({ role: "system", content: s })), ...input.messages],
      tools,
      abortSignal: input.abort,
      maxRetries: input.retries ?? 0,
    })
  }
}
```

## 2. Turbot 核心流程现状

### 2.1 SessionLoop (session_loop.cpp)

```cpp
LoopResult SessionLoop::run(const std::string& user_input) {
    // 1. 处理用户消息
    process_user_message(user_input);

    // 2. 循环执行
    while (true) {
        LoopResult result = step();
        if (result != LoopResult::Continue) {
            return result;
        }
    }
}

LoopResult SessionLoop::step() {
    // 1. 检查停止条件
    if (stop_requested_ || iteration_count_ >= config_.max_iterations) {
        return LoopResult::Stop;
    }

    // 2. 检查是否需要压缩
    if (needs_compaction()) {
        return LoopResult::Compact;
    }

    // 3. 处理 LLM 响应
    return process_llm_response();
}

LoopResult SessionLoop::process_llm_response() {
    // 1. 构建消息
    auto messages = build_llm_messages();

    // 2. 构建工具定义
    auto tools = build_tool_definitions();

    // 3. 调用 LLM
    auto stream = llm_->stream(messages, tools);

    // 4. 处理流式响应
    for (const auto& event : stream) {
        // 处理事件
    }

    // 5. 执行工具调用
    for (const auto& tc : tool_calls) {
        execute_tool(tc.name, tc.id, tc.input);
    }

    return tool_calls.empty() ? LoopResult::Stop : LoopResult::Continue;
}
```

## 4. 2.5.2 快照追踪系统

> **工时**: 1.5 天
> **测试类型**: 单元测试

### 4.1 功能说明

快照追踪用于记录 Agent 执行过程中的文件变更，支持：

1. **变更追踪**: 记录文件创建/修改/删除
2. **Patch 生成**: 生成文件变更的差异
3. **回滚支持**: 支持撤销变更

### 4.2 接口设计

```cpp
// libs/core/snapshot/snapshot.hpp

#pragma once

#include <turbot/core/common/export.hpp>
#include <string>
#include <vector>
#include <optional>
#include <filesystem>

namespace turbot::core::snapshot {

/// 文件变更类型
enum class FileChangeType {
    Created,
    Modified,
    Deleted
};

/// 文件变更记录
struct TURBOT_CORE_API FileChange {
    std::filesystem::path path;
    FileChangeType type;
    std::string old_content;    // 对于 Modified/Deleted
    std::string new_content;    // 对于 Created/Modified
    std::string hash;           // 内容哈希
};

/// Patch 结果
struct TURBOT_CORE_API PatchResult {
    std::string hash;
    std::vector<FileChange> files;
};

/// 快照管理器
class TURBOT_CORE_API SnapshotManager {
public:
    static SnapshotManager& instance();

    /// 开始追踪
    std::string start_tracking();

    /// 结束追踪并生成 patch
    PatchResult stop_tracking(const std::string& snapshot_id);

    /// 应用 patch
    bool apply_patch(const PatchResult& patch);

    /// 回滚 patch
    bool rollback_patch(const PatchResult& patch);

private:
    SnapshotManager() = default;

    struct SnapshotData {
        std::unordered_map<std::string, std::string> file_hashes;
        std::chrono::system_clock::time_point start_time;
    };

    std::unordered_map<std::string, SnapshotData> snapshots_;
};

} // namespace turbot::core::snapshot
```

### 4.3 集成到 SessionLoop

```cpp
// 在 step_start 时开始追踪
std::string snapshot_id = SnapshotManager::instance().start_tracking();

// 在 finish_step 时生成 patch
auto patch = SnapshotManager::instance().stop_tracking(snapshot_id);
if (!patch.files.empty()) {
    // 创建 patch Part
    message.add_part(Part::create_patch(patch));
}
```
