```cpp
// libs/core/session/agent_loop.hpp
#pragma once

#include "prompt.hpp"
#include <turbot/core/llm/llm.hpp>
#include <turbot/core/tool/tool.hpp>
#include <turbot/core/session/session.hpp>
#include <functional>

namespace turbot::core::session {

/// Loop 执行结果
enum class LoopResult {
    Continue,   // 继续执行
    Stop,       // 停止执行
    Error       // 错误
};

/// Agent Loop 参数
struct AgentLoopParams {
    Session session;
    agent::AgentInfo agent;
    provider::Model model;
    std::shared_ptr<std::atomic<bool>> abort_flag;
};

/// Agent Loop 执行器
class AgentLoop {
public:
    /// 创建 Loop 实例
    static AgentLoop create(const AgentLoopParams& params);

    /// 执行主循环
    LoopResult run(const std::string& user_message);

    /// 执行单步
    LoopResult step();

    /// 停止执行
    void stop();

    /// 获取当前消息
    const message::Message& current_message() const;

    /// 获取执行统计
    struct Stats {
        int total_steps = 0;
        int tool_calls = 0;
        llm::TokenUsage tokens;
    };
    Stats stats() const;

private:
    /// 处理流式事件
    LoopResult process_event(const llm::StreamEvent& event);

    /// 执行工具调用
    message::Part execute_tool(
        const std::string& tool_name,
        const nlohmann::json& input,
        const std::string& call_id
    );

    /// 死循环检测
    bool is_doom_loop(
        const std::string& tool_name,
        const nlohmann::json& input
    ) const;

    /// 更新消息 Part
    void update_part(const message::Part& part);

    Session session_;
    agent::AgentInfo agent_;
    provider::Model model_;
    message::Message assistant_message_;
    std::shared_ptr<std::atomic<bool>> abort_flag_;

    std::unordered_map<std::string, message::Part> tool_calls_;
    std::vector<std::pair<std::string, nlohmann::json>> recent_tool_calls_;

    bool stopped_ = false;
    Stats stats_;

    static constexpr int DOOM_LOOP_THRESHOLD = 3;
};

} // namespace turbot::core::session
```

```cpp
// libs/core/test/agent_loop_test.cpp

// 1. 基本文本响应测试
TEST(AgentLoop, TextResponse) {
    // Mock LLM 返回文本
    AgentLoopParams params;
    params.session = create_test_session();
    params.agent.name = "build";

    AgentLoop loop = AgentLoop::create(params);

    // Mock LLM 事件
    mock_llm_events({
        StreamEvent::TextStart{"t1"},
        StreamEvent::TextDelta{"t1", "Hello"},
        StreamEvent::TextEnd{"t1"},
        StreamEvent::Finish{}
    });

    auto result = loop.run("Say hello");
    EXPECT_EQ(result, LoopResult::Stop);
    EXPECT_FALSE(loop.current_message().parts.empty());
}

// 2. 工具调用测试
TEST(AgentLoop, ToolCall) {
    AgentLoopParams params;
    params.session = create_test_session();
    params.agent.name = "build";

    AgentLoop loop = AgentLoop::create(params);

    // Mock LLM 返回工具调用
    mock_llm_events({
        StreamEvent::ToolCall{
            "call-1",
            "read",
            {{"path", "/tmp/test.txt"}}
        },
        StreamEvent::Finish{}
    });

    auto result = loop.run("Read a file");
    EXPECT_EQ(result, LoopResult::Continue);
    EXPECT_EQ(loop.stats().tool_calls, 1);
}

// 3. 死循环检测测试
TEST(AgentLoop, DoomLoopDetection) {
    AgentLoopParams params;
    params.session = create_test_session();
    params.agent.name = "build";

    AgentLoop loop = AgentLoop::create(params);

    // Mock 连续相同工具调用
    for (int i = 0; i < 5; i++) {
        mock_llm_events({
            StreamEvent::ToolCall{
                "call-" + std::to_string(i),
                "read",
                {{"path", "/same/path"}}  // 相同参数
            },
            StreamEvent::Finish{}
        });
    }

    // 第 4 次应该触发死循环检测
    bool doom_detected = false;
    for (int i = 0; i < 5; i++) {
        loop.step();
        // 检查是否设置了 doom loop 标志
    }

    EXPECT_TRUE(doom_detected);
}

// 4. 中止测试
TEST(AgentLoop, Abort) {
    AgentLoopParams params;
    params.session = create_test_session();
    params.abort_flag = std::make_shared<std::atomic<bool>>(false);

    AgentLoop loop = AgentLoop::create(params);

    // 在执行中设置中止标志
    std::thread([&params]() {
        std::this_thread::sleep_for(std::chrono::milliseconds(100));
        params.abort_flag->store(true);
    }).detach();

    auto result = loop.run("Long running task");
    // 应该被中止
}

// 5. 统计测试
TEST(AgentLoop, Stats) {
    AgentLoopParams params;
    params.session = create_test_session();

    AgentLoop loop = AgentLoop::create(params);

    mock_llm_events({
        StreamEvent::TextDelta{"t1", "text"},
        StreamEvent::ToolCall{"c1", "read", {}},
        StreamEvent::StepFinish{"stop", {100, 50, 50, 0}},
        StreamEvent::Finish{}
    });

    loop.run("Test");

    auto stats = loop.stats();
    EXPECT_EQ(stats.tool_calls, 1);
    EXPECT_EQ(stats.tokens.total, 100);
}
```
