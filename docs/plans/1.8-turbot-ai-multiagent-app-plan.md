struct TaskParams {

class TaskTool : public core::tool::Tool {
public:
    std::string name() const override { return "task"; }

        return "Execute a subtask using a subagent";

### 2.3 子会话管理

```cpp
Task<ToolResult> TaskTool::execute(const nlohmann::json& input,
                                    ToolContext& ctx) {
    TaskParams params = input.get<TaskParams>();

    // 获取子代理
    auto agent = co_await agent::Agent::get(params.subagent_type);
    if (!agent) {
        throw std::runtime_error("Agent not found: " + params.subagent_type);
    }

    // 创建或恢复子会话
    session::Session sub_session;
    if (params.task_id) {
        auto existing = co_await session::Session::get(*params.task_id);
        if (existing) {
            sub_session = std::move(*existing);
        }
    }

    if (!sub_session) {
        sub_session = co_await session::Session::create({
            .parent_id = ctx.session_id,
            .title = params.description + " (@" + agent->name() + " subagent)",
            .permission = build_subagent_permission(*agent)
        });
    }

    // 在子会话中执行任务
    auto result = co_await execute_in_subsession(sub_session, *agent, params, ctx);

    co_return ToolResult{
        .title = params.description,
        .output = "task_id: " + sub_session.id() + "\n" + result,
        .metadata = {
            {"session_id", sub_session.id()},
            {"subagent", params.subagent_type}
        }
    };
}
```

## 3. 应用层

### 3.1 CLI应用

```cpp
// apps/turbot-cli/main.cpp

int main(int argc, char* argv[]) {
    // 初始化日志
    turbot::infra::logger::init();

    // 解析命令
    CLI::App app{"Turbot-AI CLI"};
    app.set_version_flag("--version", "0.1.0");

    // run命令
    auto run_cmd = app.add_subcommand("run", "Run interactive session");
    std::string session_id;
    run_cmd->add_option("--session", session_id, "Session ID");

    // list命令
    auto list_cmd = app.add_subcommand("list", "List sessions");

    CLI11_PARSE(app, argc, argv);

    if (*run_cmd) {
        return run_session(session_id);
    } else if (*list_cmd) {
        return list_sessions();
    }

    return 0;
}

int run_session(const std::string& session_id) {
    // 创建或恢复会话
    session::Session session;
    if (session_id.empty()) {
        session = co_await session::Session::create({
            .project_id = "default",
            .slug = "main",
            .directory = std::filesystem::current_path().string(),
            .title = "Interactive Session"
        });
    } else {
        auto result = co_await session::Session::get(session_id);
        if (!result) {
            std::cerr << "Session not found: " << session_id << std::endl;
            return 1;
        }
        session = std::move(*result);
    }

    // 运行会话循环
    SessionLoop loop(session.id());
    co_await loop.run();

    return 0;
}
```

### 3.2 Server应用

```cpp
// apps/turbot-server/main.cpp

int main(int argc, char* argv[]) {
    turbot::infra::logger::init();

    CLI::App app{"Turbot-AI Server"};
    int port = 4096;
    app.add_option("--port", port, "Server port");

    CLI11_PARSE(app, argc, argv);

    // 创建HTTP服务器
    asio::io_context io_context;
    tcp::acceptor acceptor(io_context, tcp::endpoint(tcp::v4(), port));

    std::cout << "Server listening on port " << port << std::endl;

    // 接受连接
    while (true) {
        tcp::socket socket(io_context);
        acceptor.accept(socket);

        // 处理请求
        std::thread([socket = std::move(socket)]() mutable {
            handle_request(std::move(socket));
        }).detach();
    }

    return 0;
}

void handle_request(tcp::socket socket) {
    // 解析HTTP请求
    auto request = parse_http_request(socket);

    // 处理API请求
    if (request.path == "/api/v1/sessions") {
        if (request.method == "GET") {
            list_sessions(socket, request);
        } else if (request.method == "POST") {
            create_session(socket, request);
        }
    } else if (request.path.starts_with("/api/v1/sessions/")) {
        std::string session_id = request.path.substr(19);
        if (request.method == "GET") {
            get_session(socket, session_id);
        } else if (request.method == "POST") {
            send_message(socket, session_id, request);
        }
    }
}
```
