```cpp
// websearch_tool.cpp
ToolResult WebSearchTool::execute(const json& input, const ToolContext& ctx) {
    // 1. 权限检查
    if (ctx.ask_permission) {
        auto reply = ctx.ask_permission({...});
        if (reply.is_reject()) return ToolResult::error("Permission denied");
    }

    // 2. 获取搜索配置
    std::string api_key = get_search_api_key();
    std::string api_url = get_search_api_url();

    // 3. 发起搜索请求
    auto response = http_client::post(api_url, {
        {"query", input["query"]},
        {"api_key", api_key}
    });

    // 4. 解析结果
    std::string markdown = parse_search_results(response);

    return ToolResult::ok(markdown);
}
```

## P0-2：task 工具

> **状态**：✅ 已存在（v4.0 已实现）
> **依赖**：无（但 P0-3 general agent 依赖此工具）

Turbot 缺少子任务派发工具，无法将复杂任务分解为子任务并派发给 Agent 执行。这是实现 general agent 的前置依赖。

### OpenCode 实现参考

```typescript
// OpenCode task.ts
export const TaskTool = Tool.define('task', {
  description: DESCRIPTION,
  parameters: z.object({
    agent: z.string().describe('The agent to use'),
    prompt: z.string().describe('The task description'),
  }),
  async execute(params, ctx) {

      sessionID: ctx.sessionID,
```

### 修复方向

1. 创建 `libs/core/src/tool/builtin/task_tool.cpp`
2. 实现 TaskTool：
   - 解析 agent 名称和 prompt
   - 调用 `AgentRegistry::get()` 获取 Agent
   - 调用 `Agent::execute()` 执行子任务

3. 注册到 `ToolRegistry`

### 实现骨架

```cpp
// task_tool.cpp
ToolResult TaskTool::execute(const json& input, const ToolContext& ctx) {
    // 1. 解析参数
    std::string agent_name = input["agent"];
    std::string prompt = input["prompt"];

    // 2. 获取 Agent
    auto agent = AgentRegistry::instance().get(agent_name);
    if (!agent) {
        return ToolResult::error("Agent not found: " + agent_name);
    }

    // 3. 构造执行参数
    AgentExecuteParams params;
    params.prompt = prompt;
    params.session_id = ctx.session_id;
    params.parent_message_id = ctx.message_id;

    // 4. 执行子任务
    auto result = agent->execute(params);

    // 5. 返回结果
    if (result.is_ok()) {
        return ToolResult::ok(result.get_output());
    } else {
        return ToolResult::error(result.get_error());
    }
}
```
