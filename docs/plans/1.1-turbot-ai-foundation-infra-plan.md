```cpp
void init(const std::string& log_file, const std::string& level) {
    // 创建控制台sink
    auto console_sink = std::make_shared<spdlog::sinks::stdout_color_sink_mt>();
    console_sink->set_level(spdlog::level::info);
    console_sink->set_pattern("[%Y-%m-%d %H:%M:%S.%e] [%^%l%$] [%s:%#] %v");

    std::vector<spdlog::sink_ptr> sinks{console_sink};

    // 创建文件sink
    if (!log_file.empty()) {
        auto file_sink = std::make_shared<spdlog::sinks::rotating_file_sink_mt>(
            log_file, 1024 * 1024 * 10, 3);
        file_sink->set_level(spdlog::level::trace);
        file_sink->set_pattern("[%Y-%m-%d %H:%M:%S.%e] [%l] [%t] [%s:%#] %v");
        sinks.push_back(file_sink);
    }

    // 创建logger
    auto logger = std::make_shared<spdlog::logger>("turbot", sinks.begin(), sinks.end());
    logger->set_level(level_to_spdlog_level(level));
    logger->flush_on(spdlog::level::warn);

    // 设置为默认logger
    spdlog::set_default_logger(logger);
}
```

### 4.5 事件总线

**文件**: `libs/infra/event/event_bus.hpp`, `event_bus.cpp`

**功能**:

- 发布-订阅模式
- 类型安全的事件处理
- 异步事件分发
- 事件过滤

**接口**:

```cpp
namespace turbot::infra::event {

template<typename T>
struct Event {
    std::string name;
    T data;
    int64_t timestamp;
    std::string source;
};

template<typename T>
using EventHandler = std::function<Task<void>(const Event<T>&)>;

class EventBus {
public:
    // 单例访问
    static EventBus& instance();

    // 发布事件
    template<typename T>
    Task<void> publish(const std::string& name, T data,
                        const std::string& source = "turbot");

    // 订阅事件
    template<typename T>
    std::string subscribe(const std::string& name, EventHandler<T> handler,
                           const std::string& filter = "");

    // 取消订阅
    void unsubscribe(const std::string& name, const std::string& handler_id);

    // 获取订阅者数量
    size_t subscriber_count(const std::string& name) const;

    // 列出所有事件
    std::vector<std::string> list_events() const;

private:
    EventBus() = default;

    struct HandlerEntry {
        std::string id;
        std::string filter;
        std::function<Task<void>(const std::any&)> handler;
    };

    std::unordered_map<std::string, std::vector<HandlerEntry>> handlers_;
    mutable std::shared_mutex mutex_;
};

} // namespace turbot::infra::event
```
