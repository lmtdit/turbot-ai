```cpp
// libs/core/llm/llm.hpp
#pragma once

#include "stream_event.hpp"
#include <turbot/core/session/prompt.hpp>
#include <turbot/core/provider/provider.hpp>
#include <functional>
#include <queue>

namespace turbot::core::llm {

/// 流式调用参数
struct StreamParams {
    std::string session_id;
    std::vector<session::LLMMessage> messages;
    std::vector<session::ToolDefinition> tools;
    std::optional<std::string> tool_choice;
    double temperature = 1.0;
    std::optional<double> top_p;
    std::optional<int> max_tokens;

    // 中止信号
    std::function<bool()> is_aborted;
};

/// 流式结果
class StreamResult {
public:
    /// 获取下一个事件（阻塞）
    std::optional<StreamEvent> next();

    /// 获取所有事件（阻塞直到完成）
    std::vector<StreamEvent> collect();

    /// 是否完成
    bool is_done() const;

    /// 获取最终结果
    std::string final_text() const;

    /// 获取 token 使用量
    TokenUsage usage() const;

private:
    friend class LLM;
    std::queue<StreamEvent> events_;
    bool done_ = false;
    std::string final_text_;
    TokenUsage usage_;
};

/// LLM 调用接口
class LLM {
public:
    /// 流式调用
    static StreamResult stream(
        const provider::Provider& provider,
        const provider::Model& model,
        const StreamParams& params
    );

    /// 非流式调用
    static std::string complete(
        const provider::Provider& provider,
        const provider::Model& model,
        const StreamParams& params
    );
};

} // namespace turbot::core::llm
```

## 测试验收

```cpp
// libs/core/test/llm_test.cpp

// 1. Mock Provider 测试
class MockProvider : public provider::Provider {
public:
    std::string id() const override { return "mock"; }

    // 返回模拟流式响应
    std::queue<StreamEvent> mock_events;

    void stream_response(const StreamParams& params,
                         std::function<void(StreamEvent)> callback) override {
        while (!mock_events.empty()) {
            callback(mock_events.front());
            mock_events.pop();
        }
    }
};

TEST(LLM, StreamMockText) {
    MockProvider provider;
    provider.mock_events.push(StreamEvent{StreamEvent::TextStart{"t1"}});
    provider.mock_events.push(StreamEvent{StreamEvent::TextDelta{"t1", "Hello"}});
    provider.mock_events.push(StreamEvent{StreamEvent::TextDelta{"t1", " world"}});
    provider.mock_events.push(StreamEvent{StreamEvent::TextEnd{"t1"}});
    provider.mock_events.push(StreamEvent{StreamEvent::Finish{}});

    StreamParams params;
    params.session_id = "test";

    auto result = LLM::stream(provider, {"mock-model", "mock"}, params);

    EXPECT_FALSE(result.is_done());

    auto events = result.collect();
    EXPECT_TRUE(result.is_done());
    EXPECT_EQ(events.size(), 5);
    EXPECT_EQ(result.final_text(), "Hello world");
}

// 2. 工具调用测试
TEST(LLM, StreamToolCall) {
    MockProvider provider;
    provider.mock_events.push(StreamEvent{StreamEvent::ToolCall{
        "call-1",
        "read",
        nlohmann::json{{"path", "/tmp/test.txt"}}
    }});
    provider.mock_events.push(StreamEvent{StreamEvent::Finish{}});

    StreamParams params;
    params.session_id = "test";

    auto result = LLM::stream(provider, {"mock-model", "mock"}, params);
    auto events = result.collect();

    EXPECT_EQ(events.size(), 2);
    EXPECT_EQ(events[0].type(), "tool-call");
}

// 3. 中止测试
TEST(LLM, AbortStream) {
    MockProvider provider;
    // 添加大量事件
    for (int i = 0; i < 100; i++) {
        provider.mock_events.push(StreamEvent{StreamEvent::TextDelta{"t1", "text"}});
    }

    StreamParams params;
    params.session_id = "test";
    int call_count = 0;
    params.is_aborted = [&call_count]() { return call_count++ > 5; };

    auto result = LLM::stream(provider, {"mock-model", "mock"}, params);

    // 应该在中止时停止
    int event_count = 0;
    while (auto event = result.next()) {
        event_count++;
    }
    EXPECT_LT(event_count, 100);
}

// 4. 错误处理测试
TEST(LLM, ErrorHandling) {
    MockProvider provider;
    provider.mock_events.push(StreamEvent{StreamEvent::Error{
        "API error",
        "api_error"
    }});

    StreamParams params;
    params.session_id = "test";

    auto result = LLM::stream(provider, {"mock-model", "mock"}, params);
    auto events = result.collect();

    EXPECT_EQ(events.size(), 1);
    EXPECT_EQ(events[0].type(), "error");
}
```
