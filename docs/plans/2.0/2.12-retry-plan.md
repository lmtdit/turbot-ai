实现 API 错误重试机制，支持指数退避。

## 输入

- LLM 调用错误

## 参考

```cpp
// 错误重试逻辑
function isRetryable(error) {
  if (error.status === 429) return true // Rate limit
  if (error.status >= 500) return true // Server error
  return false
}

function calculateDelay(attempt, retryAfter) {
  const base = Math.pow(2, attempt) * 1000 // 1s, 2s, 4s...
  return retryAfter ? Math.max(base, retryAfter * 1000) : base
}
```

```cpp
// libs/core/session/retry_manager.hpp
#pragma once

#include <chrono>
#include <functional>
#include <optional>

namespace turbot::core::session {

/// 重试配置
struct RetryConfig {
    int max_attempts = 5;
    int base_delay_ms = 1000;      // 基础延迟 1s
    int max_delay_ms = 60000;      // 最大延迟 60s
    double jitter_factor = 0.1;    // 抖动因子
};

/// API 错误信息
struct APIError {
    int status_code = 0;
    std::string message;
    std::optional<std::string> code;
    std::optional<int> retry_after_seconds;
};

/// 重试管理器
class RetryManager {
public:
    /// 检查错误是否可重试
    static bool is_retryable(const APIError& error);

    /// 计算重试延迟（毫秒）
    static int calculate_delay(
        int attempt,
        const RetryConfig& config = {},
        const std::optional<APIError>& error = std::nullopt
    );

    /// 执行带重试的操作
    template<typename Func>
    static auto with_retry(
        Func&& operation,
        const RetryConfig& config = {},
        std::function<bool()> is_aborted = nullptr
    ) -> decltype(operation());

    /// 等待指定时间（可被中止）
    static bool sleep(
        int milliseconds,
        std::function<bool()> is_aborted = nullptr
    );

private:
    /// 应用抖动
    static int apply_jitter(int delay_ms, double factor);
};

// 模板实现
template<typename Func>
auto RetryManager::with_retry(
    Func&& operation,
    const RetryConfig& config,
    std::function<bool()> is_aborted
) -> decltype(operation()) {
    using ResultType = decltype(operation());

    for (int attempt = 0; attempt < config.max_attempts; attempt++) {
        try {
            return operation();
        } catch (const APIError& e) {
            if (!is_retryable(e) || attempt == config.max_attempts - 1) {
                throw;
            }

            int delay = calculate_delay(attempt, config, e);
            if (sleep(delay, is_aborted)) {
                throw;  // 被中止
            }
        }
    }

    throw APIError{0, "Max retry attempts exceeded"};
}

} // namespace turbot::core::session
```

```cpp
// libs/core/test/retry_manager_test.cpp

// 1. 可重试错误测试
TEST(RetryManager, RetryableErrors) {
    EXPECT_TRUE(RetryManager::is_retryable({429, "Rate limit"}));
    EXPECT_TRUE(RetryManager::is_retryable({500, "Internal error"}));
    EXPECT_TRUE(RetryManager::is_retryable({502, "Bad gateway"}));
    EXPECT_TRUE(RetryManager::is_retryable({503, "Service unavailable"}));
}

// 2. 不可重试错误测试
TEST(RetryManager, NonRetryableErrors) {
    EXPECT_FALSE(RetryManager::is_retryable({400, "Bad request"}));
    EXPECT_FALSE(RetryManager::is_retryable({401, "Unauthorized"}));
    EXPECT_FALSE(RetryManager::is_retryable({404, "Not found"}));
}

// 3. 延迟计算测试
TEST(RetryManager, DelayCalculation) {
    RetryConfig config;

    // 指数退避: 1s, 2s, 4s, 8s...
    int d0 = RetryManager::calculate_delay(0, config);
    EXPECT_GE(d0, 900);   // ~1s with jitter
    EXPECT_LE(d0, 1100);

    int d1 = RetryManager::calculate_delay(1, config);
    EXPECT_GE(d1, 1800);  // ~2s
    EXPECT_LE(d1, 2200);

    int d2 = RetryManager::calculate_delay(2, config);
    EXPECT_GE(d2, 3600);  // ~4s
    EXPECT_LE(d2, 4400);
}

// 4. Retry-After 测试
TEST(RetryManager, RetryAfterHeader) {
    APIError error{429, "Rate limit", std::nullopt, 10};  // 10s

    int delay = RetryManager::calculate_delay(0, {}, error);
    EXPECT_GE(delay, 10000);  // 至少 10s
}

// 5. 最大延迟测试
TEST(RetryManager, MaxDelay) {
    RetryConfig config;
    config.max_delay_ms = 10000;

    // 高次尝试不应该超过最大延迟
    int delay = RetryManager::calculate_delay(10, config);
    EXPECT_LE(delay, config.max_delay_ms);
}

// 6. 重试操作测试
TEST(RetryManager, WithRetry) {
    int call_count = 0;

    auto operation = [&call_count]() -> int {
        call_count++;
        if (call_count < 3) {
            throw APIError{429, "Rate limit"};
        }
        return 42;
    };

    RetryConfig config;
    config.base_delay_ms = 10;  // 快速测试

    int result = RetryManager::with_retry(operation, config);
    EXPECT_EQ(result, 42);
    EXPECT_EQ(call_count, 3);
}

// 7. 重试次数耗尽测试
TEST(RetryManager, RetryExhausted) {
    int call_count = 0;

    auto operation = [&call_count]() -> int {
        call_count++;
        throw APIError{429, "Rate limit"};
    };

    RetryConfig config;
    config.max_attempts = 3;
    config.base_delay_ms = 10;

    EXPECT_THROW(
        RetryManager::with_retry(operation, config),
        APIError
    );
    EXPECT_EQ(call_count, 3);
}
```
