### 接口设计

```cpp
// libs/core/skill/skill.hpp
#pragma once

#include <turbot/core/common/export.hpp>
#include <nlohmann/json.hpp>
#include <string>
#include <vector>
#include <unordered_map>
#include <optional>

namespace turbot::core::skill {

/// Skill 信息
struct TURBOT_CORE_API SkillInfo {
    std::string name;           ///< Skill 标识
    std::string description;    ///< 描述
    std::string location;       ///< SKILL.md 文件路径
    std::string content;        ///< Markdown 内容

    nlohmann::json to_json() const;
    static SkillInfo from_json(const nlohmann::json& j);
};

/// Skill 配置
struct TURBOT_CORE_API SkillConfig {
    std::vector<std::string> paths;  ///< 额外搜索路径
    std::vector<std::string> urls;   ///< 远程 skill URL
};

/// Skill 管理器
class TURBOT_CORE_API SkillManager {
public:
    /// 获取单例实例
    static SkillManager& instance();

    /// 初始化，扫描所有 skill
    void initialize(const std::string& project_dir, const SkillConfig& config = {});

    /// 获取 skill
    std::optional<SkillInfo> get(const std::string& name) const;

    /// 获取所有 skill
    std::vector<SkillInfo> all() const;

    /// 获取 skill 目录列表
    std::vector<std::string> dirs() const;

    /// 刷新 skill 列表
    void refresh();

    /// 清除缓存
    void clear();

private:
    SkillManager() = default;

    /// 扫描目录
    void scan_directory(const std::string& dir, bool is_global = false);

    /// 解析 SKILL.md
    std::optional<SkillInfo> parse_skill(const std::string& path);

    std::unordered_map<std::string, SkillInfo> skills_;
    std::vector<std::string> dirs_;
    bool initialized_ = false;

    // 扫描模式
    static constexpr const char* OPENCODE_PATTERN = "skill/**/SKILL.md";
    static constexpr const char* EXTERNAL_PATTERN = "skills/**/SKILL.md";
    static constexpr const char* EXTERNAL_DIRS[] = {".claude", ".agents"};
};

} // namespace turbot::core::skill
```

// libs/core/test/skill_test.cpp

// 1. SKILL.md 解析测试
TEST(Skill, ParseSkillMd) {
    // 创建临时 skill 文件
    std::string skill_path = create_temp_skill(R"(
---
name: test-skill
description: A test skill
---

# Test Skill

This is a test skill content.
)");

    auto skill = SkillManager::instance().parse_skill(skill_path);
    ASSERT_TRUE(skill.has_value());
    EXPECT_EQ(skill->name, "test-skill");
    EXPECT_EQ(skill->description, "A test skill");
    EXPECT_TRUE(skill->content.find("test skill content") != std::string::npos);
}

// 2. 目录扫描测试
TEST(Skill, ScanDirectory) {
    std::string dir = create_skill_directory({
        {"skill-one", "First skill"},
        {"skill-two", "Second skill"},
    });

    SkillManager mgr;
    mgr.scan_directory(dir);

    auto skills = mgr.all();
    EXPECT_EQ(skills.size(), 2);
