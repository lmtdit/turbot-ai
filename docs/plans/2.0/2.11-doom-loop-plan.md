### 检测逻辑

1. 记录最近一次工具调用的名称
2. 如果连续调用相同工具，增加计数器
3. 计数器达到阈值时触发死循环检测
4. 触发时调用 on_doom_loop_detected 回调

### 对齐 OpenCode

| OpenCode                  | turbot-ai                                     |
| ------------------------- | --------------------------------------------- |
| `DOOM_LOOP_THRESHOLD = 3` | `config.doom_loop_threshold = 3`              |
| 在 processor.ts 中检测    | 在 SessionLoop::process_llm_response() 中检测 |

## 参考

```typescript
// OpenCode 死循环检测
const DOOM_LOOP_THRESHOLD = 3

function isDoomLoop(recentCalls, toolName, input) {
  let count = 0
  for (const call of recentCalls) {
    if (call.name === toolName && isEqual(call.input, input)) {
      count++
      if (count >= DOOM_LOOP_THRESHOLD) return true
    }
  }
  return false
}
```

```cpp
// libs/core/session/doom_loop_detector.hpp
#pragma once

#include <nlohmann/json.hpp>
#include <string>
#include <vector>

namespace turbot::core::session {

/// 工具调用记录
struct ToolCallRecord {
    std::string tool_name;
    nlohmann::json input;
    int64_t timestamp;
};

/// 死循环检测器
class DoomLoopDetector {
public:
    /// 检测阈值（连续相同调用次数）
    static constexpr int THRESHOLD = 3;

    /// 检测是否为死循环
    static bool is_doom_loop(
        const std::vector<ToolCallRecord>& recent_calls,
        const std::string& tool_name,
        const nlohmann::json& input
    );

    /// 添加调用记录
    void add_call(const std::string& tool_name, const nlohmann::json& input);

    /// 获取最近调用
    const std::vector<ToolCallRecord>& recent_calls() const;

    /// 清除历史
    void clear();

    /// 设置最大历史记录数
    void set_max_history(int max) { max_history_ = max; }

private:
    std::vector<ToolCallRecord> recent_calls_;
    int max_history_ = 100;

    /// 判断两次调用是否相同
    static bool is_same_call(
        const ToolCallRecord& record,
        const std::string& tool_name,
        const nlohmann::json& input
    );
};

} // namespace turbot::core::session
```

```cpp
// libs/core/test/doom_loop_detector_test.cpp

// 1. 正常调用检测测试
TEST(DoomLoopDetector, NormalCalls) {
    std::vector<ToolCallRecord> calls = {
        {"read", {{"path", "/a"}}},
        {"read", {{"path", "/b"}}},
        {"read", {{"path", "/c"}}}
    };

    EXPECT_FALSE(DoomLoopDetector::is_doom_loop(
        calls, "read", {{"path", "/d"}}
    ));
}

// 2. 死循环检测测试
TEST(DoomLoopDetector, DetectDoomLoop) {
    std::vector<ToolCallRecord> calls = {
        {"read", {{"path", "/same"}}},
        {"read", {{"path", "/same"}}},
        {"read", {{"path", "/same"}}}
    };

    EXPECT_TRUE(DoomLoopDetector::is_doom_loop(
        calls, "read", {{"path", "/same"}}
    ));
}

// 3. 不同工具测试
TEST(DoomLoopDetector, DifferentTools) {
    std::vector<ToolCallRecord> calls = {
        {"read", {{"path", "/a"}}},
        {"write", {{"path", "/a"}}},
        {"read", {{"path", "/a"}}}
    };

    // 不同工具不算死循环
    EXPECT_FALSE(DoomLoopDetector::is_doom_loop(
        calls, "read", {{"path", "/a"}}
    ));
}

// 4. 阈值边界测试
TEST(DoomLoopDetector, ThresholdBoundary) {
    std::vector<ToolCallRecord> calls;

    // 2 次相同调用，不应触发
    calls = {
        {"read", {{"path", "/same"}}},
        {"read", {{"path", "/same"}}}
    };
    EXPECT_FALSE(DoomLoopDetector::is_doom_loop(
        calls, "read", {{"path", "/same"}}
    ));

    // 3 次相同调用，应触发
    calls.push_back({"read", {{"path", "/same"}}});
    EXPECT_TRUE(DoomLoopDetector::is_doom_loop(
        calls, "read", {{"path", "/same"}}
    ));
}

// 5. 实例方法测试
TEST(DoomLoopDetector, InstanceMethods) {
    DoomLoopDetector detector;

    detector.add_call("read", {{"path", "/a"}});
    detector.add_call("read", {{"path", "/a"}});
    detector.add_call("read", {{"path", "/a"}});

    EXPECT_EQ(detector.recent_calls().size(), 3);
    EXPECT_TRUE(DoomLoopDetector::is_doom_loop(
        detector.recent_calls(), "read", {{"path", "/a"}}
    ));

    detector.clear();
    EXPECT_EQ(detector.recent_calls().size(), 0);
}

// 6. 历史限制测试
TEST(DoomLoopDetector, HistoryLimit) {
    DoomLoopDetector detector;
    detector.set_max_history(5);

    for (int i = 0; i < 10; i++) {
        detector.add_call("read", {{"path", "/file" + std::to_string(i)}});
    }

    EXPECT_EQ(detector.recent_calls().size(), 5);
}
```
