## 目标

参考 OpenCode `session/system.ts` 实现系统提示词构建。

## 输入

- AgentInfo（含 prompt 字段）
- Provider Model 信息

## 参考

```typescript
// opencode/packages/opencode/src/session/system.ts
export namespace System {
  export async function build(input: BuildInput) {
    const parts: string[] = []
    parts.push(await instructions())
    parts.push(await providerPrompt(input.model))
    if (input.agent.prompt) parts.push(input.agent.prompt)
    return parts.filter(Boolean).join('\n\n')
  }
}
```

```cpp
// libs/core/session/system_prompt.hpp
#pragma once

#include <turbot/core/agent/agent.hpp>
#include <turbot/core/provider/model.hpp>
#include <string>
#include <vector>

namespace turbot::core::session {

/// 系统提示词构建参数
struct SystemPromptParams {
    std::string session_id;
    agent::AgentInfo agent;
    provider::Model model;
    std::vector<std::string> custom_prompts;
};

/// 系统提示词构建器
class SystemPrompt {
public:
    /// 构建完整系统提示词
    static std::string build(const SystemPromptParams& params);

    /// 获取基础指令
    static std::string instructions();

    /// 获取 Provider 特定提示词
    static std::string provider_prompt(const provider::Model& model);
};

} // namespace turbot::core::session
```

## 测试验收

```cpp
// libs/core/test/system_prompt_test.cpp

// 1. 基础指令测试
TEST(SystemPrompt, Instructions) {
    auto instr = SystemPrompt::instructions();
    EXPECT_FALSE(instr.empty());
    EXPECT_TRUE(instr.find("assistant") != std::string::npos);
}

// 2. Provider 提示词测试
TEST(SystemPrompt, ProviderPrompt) {
    provider::Model anthropic{"claude-3-opus", "anthropic"};
    auto prompt = SystemPrompt::provider_prompt(anthropic);
    EXPECT_FALSE(prompt.empty());
}

// 3. 完整构建测试
TEST(SystemPrompt, Build) {
    SystemPromptParams params;
    params.session_id = "test-session";
    params.agent.name = "build";
    params.agent.prompt = "You are a coding assistant";

    auto result = SystemPrompt::build(params);
    EXPECT_FALSE(result.empty());
    // 包含基础指令
    EXPECT_TRUE(result.find("assistant") != std::string::npos);
    // 包含 Agent 提示词
    EXPECT_TRUE(result.find("coding assistant") != std::string::npos);
}

// 4. 自定义提示词测试
TEST(SystemPrompt, CustomPrompts) {
    SystemPromptParams params;
    params.agent.name = "test";
    params.custom_prompts = {"Custom instruction 1", "Custom instruction 2"};

    auto result = SystemPrompt::build(params);
    EXPECT_TRUE(result.find("Custom instruction 1") != std::string::npos);
    EXPECT_TRUE(result.find("Custom instruction 2") != std::string::npos);
}
```
