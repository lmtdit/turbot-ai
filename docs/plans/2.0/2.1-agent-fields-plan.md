// libs/core/test/agent_test.cpp

// 1. 字段读写测试
TEST(AgentInfo, NewFields) {
    AgentInfo info;
    info.prompt = "You are a helpful assistant";
    info.temperature = 0.7;
    info.top_p = 0.9;
    info.steps = 10;
    info.color = "#FF5733";

    EXPECT_EQ(info.prompt, "You are a helpful assistant");
    EXPECT_DOUBLE_EQ(*info.temperature, 0.7);
    EXPECT_DOUBLE_EQ(*info.top_p, 0.9);
    EXPECT_EQ(*info.steps, 10);
    EXPECT_EQ(info.color, "#FF5733");
}

// 2. JSON 序列化测试
TEST(AgentInfo, JsonSerialization) {
    AgentInfo info;
    info.name = "test";
    info.prompt = "test prompt";
    info.temperature = 0.5;

    auto j = info.to_json();
    EXPECT_EQ(j["name"], "test");
    EXPECT_EQ(j["prompt"], "test prompt");
    EXPECT_DOUBLE_EQ(j["temperature"].get<double>(), 0.5);

    auto parsed = AgentInfo::from_json(j);
    EXPECT_EQ(parsed.name, "test");
    EXPECT_EQ(parsed.prompt, "test prompt");
    EXPECT_DOUBLE_EQ(*parsed.temperature, 0.5);
}

// 3. 可选字段测试
TEST(AgentInfo, OptionalFields) {
    AgentInfo info;
    info.name = "minimal";

    auto j = info.to_json();
    EXPECT_FALSE(j.contains("prompt"));
    EXPECT_FALSE(j.contains("temperature"));
}
