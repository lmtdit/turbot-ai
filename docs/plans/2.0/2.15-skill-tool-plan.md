// libs/core/tool/skill_tool.cpp

std::string SkillTool::description() const {
    auto skills = skill::SkillManager::instance().all();

    if (skills.empty()) {
        return "Load a specialized skill that provides domain-specific "
               "instructions. No skills are currently available.";
    }

    std::ostringstream oss;
    oss << "Load a specialized skill that provides domain-specific "
        << "instructions and workflows.\n\n"
        << "Available skills:\n";

    for (const auto& skill : skills) {
        oss << "  - " << skill.name << ": " << skill.description << "\n";
    }

    return oss.str();
}

ToolResult SkillTool::execute(
    const nlohmann::json& input,
    const ToolContext& context
) {
    std::string skill_name = input["name"];

    auto skill = skill::SkillManager::instance().get(skill_name);
    if (!skill) {
        return ToolResult::error("Skill not found: " + skill_name);
    }

    // 权限检查
    // ...

    // 返回 skill 内容
    std::string output = generate_output(*skill);
    return ToolResult::ok(output);
}

TEST(SkillTool, PermissionFilter) {
    // 创建 skill
    create_test_skill("allowed-skill", "Allowed");
    create_test_skill("denied-skill", "Denied");

    // 设置权限
    ToolContext ctx;
    ctx.agent.permission.set_rule("skill", "denied-skill",
                                   permission::PermissionAction::Deny);

    SkillTool tool;
    auto desc = tool.description();

    // denied-skill 不应出现在描述中
    EXPECT_TRUE(desc.find("allowed-skill") != std::string::npos);
    EXPECT_TRUE(desc.find("denied-skill") == std::string::npos);
