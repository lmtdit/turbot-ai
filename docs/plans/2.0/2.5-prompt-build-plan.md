### 核心接口

```cpp
namespace turbot::core::llm {

/// Prompt 构建参数
struct PromptBuildParams {
    std::string session_id;
    agent::AgentInfo agent;
    std::string model_id;
    std::string provider_id;
    std::string working_directory;
    bool is_git_repo = false;
    std::string platform;
    std::string current_date;
    std::string user_message;
    std::vector<std::string> custom_prompts;
    std::vector<std::string> user_system_prompts;
    std::vector<std::string> allowed_tools;
    MessageFormat format = MessageFormat::OpenAI;
};

/// 工具定义
struct ToolDefinition {
    std::string name;
    std::string description;
    nlohmann::json parameters;

    nlohmann::json to_openai() const;
    nlohmann::json to_anthropic() const;
};

/// Prompt 构建结果
struct PromptBuildResult {
    std::string system;
    std::vector<LlmMessage> messages;
    std::vector<ToolDefinition> tools;

    nlohmann::json to_json() const;
    nlohmann::json build_messages_json(MessageFormat format) const;
    nlohmann::json build_tools_json(MessageFormat format) const;
};

/// Prompt 构建器
class PromptBuilder {
public:
    PromptBuildResult build(const PromptBuildParams& params);
    static std::string build_system(const PromptBuildParams& params);
    std::vector<LlmMessage> build_messages(const PromptBuildParams& params, MessageFormat format);
    std::vector<ToolDefinition> build_tools(const std::vector<std::string>& allowed_tools);
    static ToolDefinition tool_to_definition(const tool::ToolPtr& tool);
};

} // namespace turbot::core::llm
```

### 关键特性

1. **组件集成**: 整合 SystemPrompt、MessageBuilder、ToolRegistry
2. **格式支持**: OpenAI 和 Anthropic 两种输出格式
3. **工具过滤**: 支持按名称过滤可用工具
4. **空指针安全**: 添加了空指针检查
