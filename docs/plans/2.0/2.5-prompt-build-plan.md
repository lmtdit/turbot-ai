```cpp
// libs/core/session/prompt.hpp
#pragma once

#include "system_prompt.hpp"
#include "tool_schema.hpp"
#include "message_builder.hpp"
#include <turbot/core/session/session.hpp>

namespace turbot::core::session {

/// Prompt 构建参数
struct PromptBuildParams {
    std::string session_id;
    agent::AgentInfo agent;
    provider::Model model;
    std::string user_message;
    std::vector<std::string> custom_prompts;
};

/// Prompt 构建结果
struct PromptBuildResult {
    std::string system;              // 系统提示词
    std::vector<LLMMessage> messages; // 消息列表
    std::vector<ToolDefinition> tools; // 工具定义

    nlohmann::json to_json() const;
};

/// Prompt 构建器
class Prompt {
public:
    /// 构建完整 Prompt
    PromptBuildResult build(const PromptBuildParams& params);

private:
    SystemPrompt system_prompt_;
    MessageBuilder message_builder_;
    ToolSchemaBuilder tool_builder_;
};

} // namespace turbot::core::session
```

## 测试验收

```cpp
// libs/core/test/prompt_test.cpp

// 1. 完整构建测试
TEST(Prompt, BuildComplete) {
    PromptBuildParams params;
    params.session_id = "test-session";
    params.agent.name = "build";
    params.agent.prompt = "You are a coding assistant";
    params.model = provider::Model{"claude-3-opus", "anthropic"};
    params.user_message = "Write a hello world program";

    Prompt prompt_builder;
    auto result = prompt_builder.build(params);

    // 验证系统提示词
    EXPECT_FALSE(result.system.empty());
    EXPECT_TRUE(result.system.find("coding assistant") != std::string::npos);

    // 验证消息列表
    EXPECT_FALSE(result.messages.empty());

    // 验证工具定义
    EXPECT_FALSE(result.tools.empty());
}

// 2. JSON 输出测试
TEST(Prompt, JsonOutput) {
    PromptBuildParams params;
    params.session_id = "test";
    params.agent.name = "test";
    params.user_message = "Hello";

    Prompt prompt_builder;
    auto result = prompt_builder.build(params);
    auto json = result.to_json();

    EXPECT_TRUE(json.contains("system"));
    EXPECT_TRUE(json.contains("messages"));
    EXPECT_TRUE(json.contains("tools"));
}

// 3. 不同 Agent 测试
TEST(Prompt, DifferentAgents) {
    Prompt prompt_builder;

    // build agent
    PromptBuildParams build_params;
    build_params.agent.name = "build";
    build_params.agent.prompt = "Build agent prompt";
    auto build_result = prompt_builder.build(build_params);

    // plan agent
    PromptBuildParams plan_params;
    plan_params.agent.name = "plan";
    plan_params.agent.prompt = "Plan agent prompt";
    auto plan_result = prompt_builder.build(plan_params);

    // 验证不同 agent 有不同提示词
    EXPECT_TRUE(build_result.system.find("Build agent") != std::string::npos);
    EXPECT_TRUE(plan_result.system.find("Plan agent") != std::string::npos);
}

// 4. 工具过滤测试
TEST(Prompt, ToolFiltering) {
    // 创建带权限限制的 agent
    PromptBuildParams params;
    params.agent.name = "explore";
    // explore agent 只能使用 read 类工具

    Prompt prompt_builder;
    auto result = prompt_builder.build(params);

    // 验证工具列表被正确过滤
    for (const auto& tool : result.tools) {
        // explore 不应该有 write 类工具
        EXPECT_NE(tool.name, "write");
        EXPECT_NE(tool.name, "edit");
    }
}
```
