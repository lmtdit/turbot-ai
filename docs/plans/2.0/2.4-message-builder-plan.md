# 2.4 消息构建器

> **依赖**: 无（独立）
> **工时**: 1 天
> **测试**: 单元测试

## 目标

将内部消息格式转换为 LLM API 消息格式。

## 输入

- 现有 `libs/core/message/message.hpp`
- 消息历史列表

## 参考

```typescript
// OpenCode 消息格式转换
messages = history.map((msg) => ({
  role: msg.role,
  content: convertParts(msg.parts),
}))
```

### 接口设计

```cpp
// libs/core/session/message_builder.hpp
#pragma once

#include <turbot/core/message/message.hpp>
#include <nlohmann/json.hpp>
#include <vector>

namespace turbot::core::session {

/// LLM 消息格式
struct LLMMessage {
    std::string role;  // system, user, assistant
    nlohmann::json content;  // string 或 array

    nlohmann::json to_json() const;
};

/// 消息构建选项
struct MessageBuildOptions {
    int max_context_tokens = 128000;
    bool include_tool_results = true;
};

/// 消息构建器
class MessageBuilder {
public:
    /// 构建消息列表
    std::vector<LLMMessage> build(
        const std::vector<message::Message>& history,
        const MessageBuildOptions& options = {}
    );

    /// 转换单条消息
    LLMMessage convert_message(const message::Message& msg);

    /// 转换消息 Parts
    nlohmann::json convert_parts(const std::vector<message::Part>& parts);

private:
    /// 估算 token 数
    int estimate_tokens(const std::string& text) const;

    /// 截断历史
    std::vector<message::Message> truncate_history(
        const std::vector<message::Message>& history,
        int max_tokens
    );
};

} // namespace turbot::core::session
```

## 测试验收

```cpp
// libs/core/test/message_builder_test.cpp

// 1. 用户消息转换测试
TEST(MessageBuilder, UserMessage) {
    message::Message msg;
    msg.role = message::MessageRole::User;
    msg.parts.push_back(message::Part::create_text("Hello"));

    MessageBuilder builder;
    auto llm_msg = builder.convert_message(msg);

    EXPECT_EQ(llm_msg.role, "user");
    EXPECT_EQ(llm_msg.content, "Hello");
}

// 2. 助手消息转换测试
TEST(MessageBuilder, AssistantMessage) {
    message::Message msg;
    msg.role = message::MessageRole::Assistant;
    msg.parts.push_back(message::Part::create_text("Hi there!"));

    MessageBuilder builder;
    auto llm_msg = builder.convert_message(msg);

    EXPECT_EQ(llm_msg.role, "assistant");
    EXPECT_EQ(llm_msg.content, "Hi there!");
}

// 3. 工具调用消息测试
TEST(MessageBuilder, ToolCallMessage) {
    message::Message msg;
    msg.role = message::MessageRole::Assistant;
    auto tool_part = message::Part::create_tool("read", R"({"path": "/tmp/test.txt"})");
    msg.parts.push_back(tool_part);

    MessageBuilder builder;
    auto llm_msg = builder.convert_message(msg);

    EXPECT_EQ(llm_msg.role, "assistant");
    EXPECT_TRUE(llm_msg.content.is_array());
    // 验证 tool_calls 格式
}

// 4. 历史截断测试
TEST(MessageBuilder, TruncateHistory) {
    std::vector<message::Message> history;
    // 添加大量消息
    for (int i = 0; i < 100; i++) {
        message::Message msg;
        msg.role = message::MessageRole::User;
        msg.parts.push_back(message::Part::create_text("Message " + std::to_string(i)));
        history.push_back(msg);
    }

    MessageBuilder builder;
    MessageBuildOptions options;
    options.max_context_tokens = 1000;  // 小限制

    auto result = builder.build(history, options);
    // 验证消息被截断
    EXPECT_LT(result.size(), 100);
}

// 5. 完整构建测试
TEST(MessageBuilder, BuildComplete) {
    std::vector<message::Message> history;

    message::Message user_msg;
    user_msg.role = message::MessageRole::User;
    user_msg.parts.push_back(message::Part::create_text("Hello"));
    history.push_back(user_msg);

    message::Message assistant_msg;
    assistant_msg.role = message::MessageRole::Assistant;
    assistant_msg.parts.push_back(message::Part::create_text("Hi!"));
    history.push_back(assistant_msg);

    MessageBuilder builder;
    auto messages = builder.build(history);

    EXPECT_EQ(messages.size(), 2);
    EXPECT_EQ(messages[0].role, "user");
    EXPECT_EQ(messages[1].role, "assistant");
}
```
