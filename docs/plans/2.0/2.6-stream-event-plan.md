```cpp
// libs/core/llm/stream_event.hpp
#pragma once

#include <nlohmann/json.hpp>
#include <string>
#include <variant>
#include <optional>

namespace turbot::core::llm {

/// Token 使用量
struct TokenUsage {
    int total = 0;
    int input = 0;
    int output = 0;
    int reasoning = 0;

    struct Cache {
        int read = 0;
        int write = 0;
    } cache;

    nlohmann::json to_json() const;
    static TokenUsage from_json(const nlohmann::json& j);
};

/// 流式事件类型
struct StreamEvent {
    // 文本开始
    struct TextStart {
        std::string id;
    };

    // 文本增量
    struct TextDelta {
        std::string id;
        std::string text;
    };

    // 文本结束
    struct TextEnd {
        std::string id;
    };

    // 工具输入开始
    struct ToolInputStart {
        std::string id;
        std::string tool_name;
    };

    // 工具输入增量
    struct ToolInputDelta {
        std::string id;
        std::string delta;
    };

    // 工具调用完成
    struct ToolCall {
        std::string id;
        std::string tool_name;
        nlohmann::json input;
    };

    // 推理开始
    struct ReasoningStart {
        std::string id;
    };

    // 推理增量
    struct ReasoningDelta {
        std::string id;
        std::string text;
    };

    // 推理结束
    struct ReasoningEnd {
        std::string id;
    };

    // 步骤开始
    struct StepStart {};

    // 步骤结束
    struct StepFinish {
        std::string finish_reason;
        TokenUsage usage;
    };

    // 错误
    struct Error {
        std::string message;
        std::optional<std::string> code;
    };

    // 流结束
    struct Finish {};

    // 事件数据
    std::variant<
        TextStart, TextDelta, TextEnd,
        ToolInputStart, ToolInputDelta, ToolCall,
        ReasoningStart, ReasoningDelta, ReasoningEnd,
        StepStart, StepFinish, Error, Finish
    > data;

    // 获取事件类型名
    std::string type() const;

    // JSON 序列化
    nlohmann::json to_json() const;
};

// 辅助函数
std::string event_type_name(const StreamEvent& event);

} // namespace turbot::core::llm
```

// libs/core/test/stream_event_test.cpp

// 1. TextDelta 测试
TEST(StreamEvent, TextDelta) {
    StreamEvent event;
    event.data = StreamEvent::TextDelta{"id-1", "Hello"};

    EXPECT_EQ(event.type(), "text-delta");
    EXPECT_EQ(event.to_json()["type"], "text-delta");
    EXPECT_EQ(event.to_json()["id"], "id-1");
    EXPECT_EQ(event.to_json()["text"], "Hello");
}

// 2. ToolCall 测试
TEST(StreamEvent, ToolCall) {
    StreamEvent event;
    event.data = StreamEvent::ToolCall{
        "call-1",
        "read",
        nlohmann::json{{"path", "/tmp/test.txt"}}
    };

    EXPECT_EQ(event.type(), "tool-call");
    auto j = event.to_json();
    EXPECT_EQ(j["tool_name"], "read");
    EXPECT_EQ(j["input"]["path"], "/tmp/test.txt");
}

// 3. StepFinish 测试
TEST(StreamEvent, StepFinish) {
    StreamEvent event;
    event.data = StreamEvent::StepFinish{
        "stop",
        TokenUsage{100, 50, 50, 0}
    };

    EXPECT_EQ(event.type(), "finish-step");
    auto j = event.to_json();
    EXPECT_EQ(j["finish_reason"], "stop");
    EXPECT_EQ(j["usage"]["total"], 100);
}

// 4. Error 测试
TEST(StreamEvent, Error) {
    StreamEvent event;
    event.data = StreamEvent::Error{
        "Rate limit exceeded",
        "rate_limit"
    };

    EXPECT_EQ(event.type(), "error");
    auto j = event.to_json();
    EXPECT_EQ(j["message"], "Rate limit exceeded");
    EXPECT_EQ(j["code"], "rate_limit");
}

// 5. TokenUsage 测试
TEST(TokenUsage, JsonRoundtrip) {
    TokenUsage usage;
    usage.total = 1000;
    usage.input = 600;
    usage.output = 400;
    usage.reasoning = 100;
    usage.cache.read = 200;
    usage.cache.write = 100;

    auto j = usage.to_json();
    auto parsed = TokenUsage::from_json(j);

    EXPECT_EQ(parsed.total, 1000);
    EXPECT_EQ(parsed.input, 600);
    EXPECT_EQ(parsed.output, 400);
    EXPECT_EQ(parsed.reasoning, 100);
    EXPECT_EQ(parsed.cache.read, 200);
    EXPECT_EQ(parsed.cache.write, 100);
}
