```cpp
// libs/core/session/usage_tracker.hpp
#pragma once

#include <turbot/core/llm/stream_event.hpp>
#include <turbot/core/provider/model.hpp>
#include <unordered_map>

namespace turbot::core::session {

/// 成本信息
struct CostInfo {
    double input_cost = 0;
    double output_cost = 0;
    double cache_read_cost = 0;
    double cache_write_cost = 0;
    double reasoning_cost = 0;
    double total_cost = 0;
};

/// 会话使用统计
struct SessionUsage {
    llm::TokenUsage tokens;
    double total_cost = 0;
    int message_count = 0;
    int tool_calls = 0;
};

/// 使用量追踪器
class UsageTracker {
public:
    /// 计算 Token 使用量
    static llm::TokenUsage calculate_usage(
        const provider::Model& model,
        const nlohmann::json& raw_usage,
        const std::optional<nlohmann::json>& metadata = std::nullopt
    );

    /// 计算成本
    static CostInfo calculate_cost(
        const provider::Model& model,
        const llm::TokenUsage& usage
    );

    /// 累加会话使用量
    void add_usage(const std::string& session_id, const llm::TokenUsage& usage);

    /// 获取会话使用量
    SessionUsage get_session_usage(const std::string& session_id) const;

    /// 获取总使用量
    SessionUsage get_total_usage() const;

    /// 重置会话使用量
    void reset_session(const std::string& session_id);

private:
    std::unordered_map<std::string, SessionUsage> sessions_;
    SessionUsage total_;
};

} // namespace turbot::core::session
```

## 测试验收

```cpp
// libs/core/test/usage_tracker_test.cpp

// 1. Token 计算测试
TEST(UsageTracker, CalculateTokens) {
    nlohmann::json raw_usage = {
        {"promptTokens", 100},
        {"completionTokens", 50}
    };

    provider::Model model{"claude-3-opus", "anthropic"};
    auto usage = UsageTracker::calculate_usage(model, raw_usage);

    EXPECT_EQ(usage.input, 100);
    EXPECT_EQ(usage.output, 50);
    EXPECT_EQ(usage.total, 150);
}

// 2. 成本计算测试
TEST(UsageTracker, CalculateCost) {
    llm::TokenUsage usage;
    usage.input = 1000;
    usage.output = 500;

    provider::Model model{"claude-3-opus", "anthropic"};
    auto cost = UsageTracker::calculate_cost(model, usage);

    // 根据定价计算（假设 $0.015/1K input, $0.075/1K output）
    EXPECT_GT(cost.input_cost, 0);
    EXPECT_GT(cost.output_cost, 0);
    EXPECT_GT(cost.total_cost, 0);
}

// 3. 缓存 Token 测试
TEST(UsageTracker, CacheTokens) {
    llm::TokenUsage usage;
    usage.input = 1000;
    usage.cache.read = 500;
    usage.cache.write = 200;

    provider::Model model{"claude-3-opus", "anthropic"};
    auto cost = UsageTracker::calculate_cost(model, usage);

    // 缓存读取成本应低于普通输入成本
    EXPECT_GT(cost.cache_read_cost, 0);
    EXPECT_LT(cost.cache_read_cost, cost.input_cost);
}

// 4. 会话累计测试
TEST(UsageTracker, SessionAccumulate) {
    UsageTracker tracker;

    tracker.add_usage("session-1", {100, 50, 50, 0});
    tracker.add_usage("session-1", {200, 100, 100, 0});

    auto usage = tracker.get_session_usage("session-1");
    EXPECT_EQ(usage.tokens.total, 450);
    EXPECT_EQ(usage.message_count, 2);
}

// 5. 总使用量测试
TEST(UsageTracker, TotalUsage) {
    UsageTracker tracker;

    tracker.add_usage("session-1", {100, 50, 50, 0});
    tracker.add_usage("session-2", {200, 100, 100, 0});

    auto total = tracker.get_total_usage();
    EXPECT_EQ(total.tokens.total, 650);
}

// 6. 重置测试
TEST(UsageTracker, ResetSession) {
    UsageTracker tracker;

    tracker.add_usage("session-1", {100, 50, 50, 0});
    tracker.reset_session("session-1");

    auto usage = tracker.get_session_usage("session-1");
    EXPECT_EQ(usage.tokens.total, 0);
    EXPECT_EQ(usage.message_count, 0);
}
```
