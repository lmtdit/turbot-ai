## 实现

### 修改文件

```cpp
// libs/core/message/part.hpp

/// Part 类型枚举
enum class PartType {
    Text,        // 文本内容
    Tool,        // 工具调用
    Reasoning,   // 推理过程
    StepStart,   // 步骤开始
    StepFinish,  // 步骤结束
    Patch,       // 代码变更
    Image,       // 图片
    Error        // 错误信息
};

/// Part 扩展数据
struct PartData {
    // 文本
    struct Text {
        std::string text;
    };

    // 工具调用
    struct Tool {
        std::string name;
        std::string call_id;
        nlohmann::json input;
        nlohmann::json output;
        std::optional<std::string> error;
        enum Status { Pending, Running, Completed, Error } status;
    };

    // 推理
    struct Reasoning {
        std::string text;
    };

    // 步骤
    struct StepStart {
        std::string snapshot_id;
    };

    struct StepFinish {
        std::string finish_reason;
        llm::TokenUsage tokens;
    };

    // 代码变更
    struct Patch {
        std::string hash;
        std::vector<FileDiff> files;
    };

    // 图片
    struct Image {
        std::string url;
        std::string media_type;
        std::vector<uint8_t> data;
    };

    // 错误
    struct Error {
        std::string message;
        std::optional<std::string> code;
    };

    std::variant<Text, Tool, Reasoning, StepStart, StepFinish, Patch, Image, Error> data;
};
```

## 测试验收

```cpp
// libs/core/test/part_test.cpp

// 1. Part 类型转换测试
TEST(Part, TypeConversion) {
    Part text_part = Part::create_text("Hello");
    EXPECT_EQ(text_part.type, PartType::Text);

    Part tool_part = Part::create_tool("read", R"({"path": "/tmp"})");
    EXPECT_EQ(tool_part.type, PartType::Tool);
}

// 2. 工具 Part 状态测试
TEST(Part, ToolStatus) {
    auto part = Part::create_tool("read", "{}");
    auto& tool = std::get<PartData::Tool>(part.data);

    EXPECT_EQ(tool.status, PartData::Tool::Pending);

    tool.status = PartData::Tool::Running;
    EXPECT_EQ(tool.status, PartData::Tool::Running);

    tool.output = {{"result", "file content"}};
    tool.status = PartData::Tool::Completed;
    EXPECT_EQ(tool.status, PartData::Tool::Completed);
}

// 3. 流式文本增量测试
TEST(Part, TextIncremental) {
    Part part = Part::create_text("");
    auto& text = std::get<PartData::Text>(part.data);

    text.text += "Hello";
    text.text += " ";
    text.text += "World";

    EXPECT_EQ(text.text, "Hello World");
}

// 4. 步骤 Part 测试
TEST(Part, StepParts) {
    Part start = Part::create_step_start("snapshot-123");
    EXPECT_EQ(start.type, PartType::StepStart);
    auto& start_data = std::get<PartData::StepStart>(start.data);
    EXPECT_EQ(start_data.snapshot_id, "snapshot-123");

    Part finish = Part::create_step_finish("stop", {100, 50, 50, 0});
    EXPECT_EQ(finish.type, PartType::StepFinish);
    auto& finish_data = std::get<PartData::StepFinish>(finish.data);
    EXPECT_EQ(finish_data.finish_reason, "stop");
    EXPECT_EQ(finish_data.tokens.total, 100);
}

// 5. JSON 序列化测试
TEST(Part, JsonSerialization) {
    Part part = Part::create_tool("read", R"({"path": "/tmp/test.txt"})");
    auto j = part.to_json();

    EXPECT_EQ(j["type"], "tool");
    EXPECT_EQ(j["name"], "read");
    EXPECT_TRUE(j.contains("input"));

    Part parsed = Part::from_json(j);
    EXPECT_EQ(parsed.type, PartType::Tool);
}

// 6. 错误 Part 测试
TEST(Part, ErrorPart) {
    Part part = Part::create_error("Something went wrong", "internal_error");
    EXPECT_EQ(part.type, PartType::Error);

    auto& error = std::get<PartData::Error>(part.data);
    EXPECT_EQ(error.message, "Something went wrong");
    EXPECT_EQ(error.code, "internal_error");
}
```
