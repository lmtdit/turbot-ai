```cpp
async function compact(sessionId, messageId) {
  const messages = await getMessages(sessionId)
  const summary = await generateSummary(messages)
  const retained = selectRetained(messages)

  // 创建压缩消息
  await createCompactionMessage(sessionId, {

    retained: retained.map((m) => m.id),
```

```cpp
// libs/core/session/session_compaction.hpp
#pragma once

#include <turbot/core/session/session.hpp>
#include <turbot/core/message/message.hpp>
#include <turbot/core/provider/model.hpp>
#include <vector>

namespace turbot::core::session {

/// 压缩配置
struct CompactionConfig {
    double target_ratio = 0.5;    // 目标压缩到 50%
    int min_messages_to_keep = 5; // 最少保留消息数
    bool include_tool_results = true;
};

/// 压缩结果
struct CompactionResult {
    std::string summary;              // 摘要内容
    std::vector<std::string> retained_ids;  // 保留的消息 ID
    int original_tokens = 0;          // 原始 token 数
    int compressed_tokens = 0;        // 压缩后 token 数
};

/// 会话压缩器
class SessionCompaction {
public:
    /// 检查是否需要压缩
    static bool is_overflow(
        const llm::TokenUsage& usage,
        const provider::Model& model
    );

    /// 执行压缩
    CompactionResult compact(
        const std::string& session_id,
        const std::vector<message::Message>& messages,
        const CompactionConfig& config = {}
    );

    /// 生成摘要
    std::string generate_summary(
        const std::vector<message::Message>& messages
    );

    /// 选择保留消息
    std::vector<message::Message> select_retained(
        const std::vector<message::Message>& messages,
        const CompactionConfig& config
    );

private:
    /// 估算消息 token 数
    int estimate_tokens(const message::Message& msg);

    /// 判断消息是否重要
    bool is_important(const message::Message& msg);
};

} // namespace turbot::core::session
```

```cpp
// libs/core/test/session_compaction_test.cpp

// 1. 溢出检测测试
TEST(SessionCompaction, OverflowDetection) {
    provider::Model model{"claude-3-opus", "anthropic"};
    model.context_window = 100000;

    llm::TokenUsage usage;
    usage.total = 50000;
    EXPECT_FALSE(SessionCompaction::is_overflow(usage, model));

    usage.total = 95000;  // 95% 使用
    EXPECT_TRUE(SessionCompaction::is_overflow(usage, model));
}

// 2. 摘要生成测试
TEST(SessionCompaction, GenerateSummary) {
    std::vector<message::Message> messages;
    messages.push_back(create_message("User", "What is AI?"));
    messages.push_back(create_message("Assistant", "AI stands for..."));
    messages.push_back(create_message("User", "Tell me more"));
    messages.push_back(create_message("Assistant", "AI includes..."));

    SessionCompaction compactor;
    std::string summary = compactor.generate_summary(messages);

    EXPECT_FALSE(summary.empty());
    // 摘要应该比原始内容短
    int original_len = 0;
    for (const auto& m : messages) {
        original_len += m.text_content().length();
    }
    EXPECT_LT(summary.length(), original_len);
}

// 3. 保留消息选择测试
TEST(SessionCompaction, SelectRetained) {
    std::vector<message::Message> messages;
    // 添加 20 条消息
    for (int i = 0; i < 20; i++) {
        messages.push_back(create_message(
            i % 2 == 0 ? "User" : "Assistant",
            "Message " + std::to_string(i)
        ));
    }

    SessionCompaction compactor;
    CompactionConfig config;
    config.min_messages_to_keep = 5;

    auto retained = compactor.select_retained(messages, config);

    // 应该保留最少消息数
    EXPECT_GE(retained.size(), config.min_messages_to_keep);
    // 应该少于原始消息数
    EXPECT_LT(retained.size(), messages.size());
}

// 4. 完整压缩测试
TEST(SessionCompaction, FullCompaction) {
    std::vector<message::Message> messages;
    for (int i = 0; i < 50; i++) {
        messages.push_back(create_message(
            i % 2 == 0 ? "User" : "Assistant",
            "This is a longer message with more content. " +
            std::string(100, 'x') + std::to_string(i)
        ));
    }

    SessionCompaction compactor;
    CompactionConfig config;
    config.target_ratio = 0.3;

    auto result = compactor.compact("test-session", messages, config);

    EXPECT_FALSE(result.summary.empty());
    EXPECT_GT(result.original_tokens, 0);
    EXPECT_LT(result.compressed_tokens, result.original_tokens);
    EXPECT_LT(result.retained_ids.size(), messages.size());
}

// 5. 重要消息保留测试
TEST(SessionCompaction, ImportantMessagesRetained) {
    std::vector<message::Message> messages;

    // 添加普通消息
    for (int i = 0; i < 10; i++) {
        messages.push_back(create_message("User", "Normal message"));
    }

    // 添加包含重要信息的消息（如工具调用结果）
    message::Message important;
    important.role = message::MessageRole::Assistant;
    important.parts.push_back(message::Part::create_tool("read", "{}"));
    messages.push_back(important);

    SessionCompaction compactor;
    auto retained = compactor.select_retained(messages, {});

    // 重要消息应该被保留
    bool has_tool_call = false;
    for (const auto& m : retained) {
        for (const auto& p : m.parts) {
            if (p.type == message::PartType::Tool) {
                has_tool_call = true;
            }
        }
    }
    EXPECT_TRUE(has_tool_call);
}

// 6. Token 估算测试
TEST(SessionCompaction, TokenEstimation) {
    SessionCompaction compactor;

    message::Message msg;
    msg.role = message::MessageRole::User;
    msg.parts.push_back(message::Part::create_text("Hello world"));

    int tokens = compactor.estimate_tokens(msg);
    // 简单估算：大约 4 字符 = 1 token
    EXPECT_GT(tokens, 0);
    EXPECT_LT(tokens, 100);
}
```
