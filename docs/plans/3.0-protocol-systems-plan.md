## 1. MCP (Model Context Protocol) 系统

### 1.1 协议概述

MCP 是 Anthropic 主导的开放协议，用于 AI 模型与外部工具/服务的标准化通信。

**核心概念**:
- **Tools**: 可调用的函数
- **Resources**: 可读取的资源
- **Prompts**: 预定义的提示词模板
- **Transports**: 通信传输层 (stdio/SSE/HTTP)

### 1.2 OpenCode 实现分析

```typescript
// opencode/packages/opencode/src/mcp/index.ts

export namespace MCP {
  // 状态类型
  export const Status = z.discriminatedUnion("status", [
    { status: "connected" },
    { status: "disabled" },
    { status: "failed", error: string },
    { status: "needs_auth" },
    { status: "needs_client_registration", error: string },
  ])
  
  // 资源定义
  export const Resource = z.object({
    name: z.string(),
    uri: z.string(),
    description: z.string().optional(),
    mimeType: z.string().optional(),
    client: z.string(),
  })
  
  // 传输类型
  type Transport = 
    | StdioClientTransport    // 本地进程通信
    | SSEClientTransport      // Server-Sent Events
    | StreamableHTTPClientTransport  // HTTP 流式
}
```

```cpp
// libs/core/mcp/mcp.hpp
#pragma once

#include <turbot/core/common/export.hpp>
#include <nlohmann/json.hpp>
#include <string>
#include <vector>
#include <functional>
#include <future>

namespace turbot::core::mcp {

/// MCP 状态
enum class MCPStatus {
    Connected,
    Disabled,
    Failed,
    NeedsAuth,
    NeedsClientRegistration,
};

/// MCP 资源定义
struct TURBOT_CORE_API MCPResource {
    std::string name;
    std::string uri;
    std::optional<std::string> description;
    std::optional<std::string> mime_type;
    std::string client;
    
    nlohmann::json to_json() const;
    static MCPResource from_json(const nlohmann::json& j);
};

/// MCP 工具定义
struct TURBOT_CORE_API MCPTool {
    std::string name;
    std::string description;
    nlohmann::json input_schema;
    
    nlohmann::json to_json() const;
    static MCPTool from_json(const nlohmann::json& j);
};

/// MCP 提示词模板
struct TURBOT_CORE_API MCPPrompt {
    std::string name;
    std::string description;
    std::vector<std::string> arguments;
    
    nlohmann::json to_json() const;
    static MCPPrompt from_json(const nlohmann::json& j);
};

/// MCP 客户端配置
struct TURBOT_CORE_API MCPClientConfig {
    std::string name;
    std::string type;  // "stdio", "sse", "http"
    
    // stdio 配置
    std::optional<std::string> command;
    std::vector<std::string> args;
    std::unordered_map<std::string, std::string> env;
    
    // remote 配置
    std::optional<std::string> url;
    std::optional<nlohmann::json> oauth;
    
    bool enabled = true;
    int timeout = 30000;  // ms
};

/// MCP 客户端
class TURBOT_CORE_API MCPClient {
public:
    virtual ~MCPClient() = default;
    
    /// 连接到 MCP 服务器
    virtual std::future<void> connect() = 0;
    
    /// 断开连接
    virtual std::future<void> disconnect() = 0;
    
    /// 获取状态
    virtual MCPStatus status() const = 0;
    
    /// 列出工具
    virtual std::future<std::vector<MCPTool>> list_tools() = 0;
    
    /// 调用工具
    virtual std::future<nlohmann::json> call_tool(
        const std::string& name,
        const nlohmann::json& arguments
    ) = 0;
    
    /// 列出资源
    virtual std::future<std::vector<MCPResource>> list_resources() = 0;
    
    /// 读取资源
    virtual std::future<nlohmann::json> read_resource(const std::string& uri) = 0;
    
    /// 列出提示词模板
    virtual std::future<std::vector<MCPPrompt>> list_prompts() = 0;
    
    /// 获取提示词
    virtual std::future<std::string> get_prompt(
        const std::string& name,
        const nlohmann::json& arguments
    ) = 0;
    
    /// 注册工具变更回调
    virtual void on_tools_changed(std::function<void()> callback) = 0;
};

/// MCP 管理器
class TURBOT_CORE_API MCPManager {
public:
    static MCPManager& instance();
    
    /// 初始化所有 MCP 服务器
    void initialize(const std::vector<MCPClientConfig>& configs);
    
    /// 添加 MCP 服务器
    std::future<MCPStatus> add(const std::string& name, const MCPClientConfig& config);
    
    /// 移除 MCP 服务器
    void remove(const std::string& name);
    
    /// 获取所有工具
    std::future<std::vector<MCPTool>> all_tools();
    
    /// 获取所有资源
    std::future<std::vector<MCPResource>> all_resources();
    
    /// 获取状态
    std::unordered_map<std::string, MCPStatus> status() const;
    
private:
    MCPManager() = default;
    
    std::unordered_map<std::string, std::unique_ptr<MCPClient>> clients_;
    std::unordered_map<std::string, MCPStatus> status_;
};

} // namespace turbot::core::mcp
```

## 2. LSP (Language Server Protocol) 系统

### 2.1 协议概述

LSP 是微软主导的协议，用于编辑器与语言服务器之间的通信，提供代码补全、诊断、跳转等功能。

### 2.2 OpenCode 实现分析

```typescript
// opencode/packages/opencode/src/lsp/index.ts

export namespace LSP {
  // 符号定义
  export const Symbol = z.object({
    name: z.string(),
    kind: z.number(),
    location: z.object({
      uri: z.string(),
      range: Range,
    }),
  })
  
  // 文档符号
  export const DocumentSymbol = z.object({
    name: z.string(),
    detail: z.string().optional(),
    kind: z.number(),
    range: Range,
    selectionRange: Range,
  })
  
  // 核心功能
  async function hover(input: { file, line, character })
  async function definition(input: { file, line, character })
  async function references(input: { file, line, character })
  async function implementation(input: { file, line, character })
  async function workspaceSymbol(query: string)
  async function documentSymbol(uri: string)
  async function diagnostics()
}
```

```cpp
// libs/core/lsp/lsp.hpp
#pragma once

#include <turbot/core/common/export.hpp>
#include <nlohmann/json.hpp>
#include <string>
#include <vector>
#include <optional>
#include <functional>

namespace turbot::core::lsp {

/// 位置
struct Position {
    int line;
    int character;
};

/// 范围
struct Range {
    Position start;
    Position end;
};

/// 位置（带 URI）
struct Location {
    std::string uri;
    Range range;
};

/// 符号
struct Symbol {
    std::string name;
    int kind;
    Location location;
};

/// 文档符号
struct DocumentSymbol {
    std::string name;
    std::optional<std::string> detail;
    int kind;
    Range range;
    Range selection_range;
    std::vector<DocumentSymbol> children;
};

/// 诊断
struct Diagnostic {
    Range range;
    int severity;  // 1=Error, 2=Warning, 3=Info, 4=Hint
    std::string message;
    std::optional<std::string> source;
    std::optional<std::string> code;
};

/// 悬停信息
struct Hover {
    std::string contents;
    std::optional<Range> range;
};

/// LSP 客户端接口
class TURBOT_CORE_API LSPClient {
public:
    virtual ~LSPClient() = default;
    
    /// 初始化
    virtual std::future<void> initialize(const std::string& root_path) = 0;
    
    /// 关闭
    virtual std::future<void> shutdown() = 0;
    
    /// 打开文件
    virtual void did_open(const std::string& uri, const std::string& language_id, const std::string& content) = 0;
    
    /// 关闭文件
    virtual void did_close(const std::string& uri) = 0;
    
    /// 文件变更
    virtual void did_change(const std::string& uri, const std::string& content) = 0;
    
    /// 获取悬停信息
    virtual std::future<std::optional<Hover>> hover(const std::string& uri, Position position) = 0;
    
    /// 跳转定义
    virtual std::future<std::vector<Location>> definition(const std::string& uri, Position position) = 0;
    
    /// 查找引用
    virtual std::future<std::vector<Location>> references(const std::string& uri, Position position) = 0;
    
    /// 查找实现
    virtual std::future<std::vector<Location>> implementation(const std::string& uri, Position position) = 0;
    
    /// 工作区符号
    virtual std::future<std::vector<Symbol>> workspace_symbol(const std::string& query) = 0;
    
    /// 文档符号
    virtual std::future<std::vector<DocumentSymbol>> document_symbol(const std::string& uri) = 0;
    
    /// 获取诊断
    virtual std::unordered_map<std::string, std::vector<Diagnostic>> diagnostics() const = 0;
    
    /// 等待诊断
    virtual std::future<void> wait_for_diagnostics(const std::string& uri) = 0;
};

/// LSP 服务器信息
struct LSPServerInfo {
    std::string id;
    std::vector<std::string> extensions;
    bool global = false;
    
    /// 根目录检测函数
    std::function<std::future<std::optional<std::string>>(const std::string& file)> root;
    
    /// 启动服务器
    std::function<std::future<std::unique_ptr<LSPClient>>(const std::string& root)> spawn;
};

/// LSP 管理器
class TURBOT_CORE_API LSPManager {
public:
    static LSPManager& instance();
    
    /// 初始化
    void initialize();
    
    /// 获取文件对应的客户端
    std::future<std::vector<LSPClient*>> get_clients(const std::string& file);
    
    /// 触摸文件（激活 LSP）
    void touch_file(const std::string& file, bool wait_for_diagnostics = false);
    
    /// 获取所有诊断
    std::unordered_map<std::string, std::vector<Diagnostic>> diagnostics();
    
    /// 悬停
    std::future<std::optional<Hover>> hover(const std::string& file, Position position);
    
    /// 定义
    std::future<std::vector<Location>> definition(const std::string& file, Position position);
    
    /// 引用
    std::future<std::vector<Location>> references(const std::string& file, Position position);
    
    /// 注册服务器
    void register_server(LSPServerInfo server);
    
    /// 获取状态
    std::vector<nlohmann::json> status() const;
    
private:
    LSPManager() = default;
    
    std::unordered_map<std::string, LSPServerInfo> servers_;
    std::vector<std::unique_ptr<LSPClient>> clients_;
    std::set<std::string> broken_;
};

} // namespace turbot::core::lsp
```

## 3. ACP (Agent Client Protocol) 系统

### 3.1 协议概述

ACP 是标准化 AI 编程智能体与 IDE 之间通信的开放协议，由 Zed 等项目推动。

**核心能力**:
- 会话管理 (new/load/resume)
- 提示处理
- 文件操作
- 权限管理
- 工具调用报告

### 3.2 OpenCode 实现分析

```typescript
// opencode/packages/opencode/src/acp/agent.ts

  // Agent 接口实现
  async initialize(request: InitializeRequest): Promise<InitializeResponse>

  async loadSession(request: LoadSessionRequest): Promise<SessionInfo>

  async cancel(notification: CancelNotification): Promise<void>
```

```cpp
// libs/core/acp/acp.hpp
#pragma once

#include <turbot/core/common/export.hpp>
#include <nlohmann/json.hpp>
#include <string>
#include <vector>
#include <optional>
#include <functional>

namespace turbot::core::acp {

/// ACP 状态类型
enum class ACPStatus {
    Connected,
    Error,
};

/// 位置
struct Position {
    int line;
    int character;
};

/// 范围
struct Range {
    Position start;
    Position end;
};

/// 文本内容块
struct TextContent {
    std::string text;
};

/// 资源内容块
struct ResourceContent {
    std::string uri;
    std::optional<std::string> mime_type;
};

/// 工具调用内容
struct ToolCallContent {
    std::string tool_kind;
    std::string tool_name;
    std::string id;
    nlohmann::json input;
    std::optional<std::string> status;
    std::optional<std::string> output;
};

/// 使用统计
struct Usage {
    int input_tokens;
    int output_tokens;
    std::optional<int> cache_write_tokens;
    std::optional<int> cache_read_tokens;
};

/// 停止原因
enum class StopReason {
    EndTurn,
    ToolUse,
    StopSequence,
    Error,
    Cancelled,
};

/// 会话信息
struct SessionInfo {
    std::string session_id;
    std::string cwd;
    std::optional<std::string> model_id;
    std::optional<std::string> mode;
    std::vector<nlohmann::json> mcp_servers;
};

/// 权限选项
struct PermissionOption {
    std::string permission;
    std::vector<std::string> patterns;
    std::optional<std::string> message;
};

/// 初始化请求
struct InitializeRequest {
    int protocol_version;
    std::optional<std::string> authentication;
};

/// 初始化响应
struct InitializeResponse {
    int protocol_version;
    std::vector<std::string> agent_capabilities;
    std::vector<std::string> auth_methods;
    std::vector<ModelOption> models;
    std::vector<ModeOption> modes;
};

/// 提示请求
struct PromptRequest {
    std::string session_id;
    std::vector<nlohmann::json> content;  // TextContent | ResourceContent
    std::optional<std::string> mode;
    std::optional<std::string> model;
};

/// ACP Agent 接口
class TURBOT_CORE_API ACPAgent {
public:
    virtual ~ACPAgent() = default;
    
    /// 初始化
    virtual InitializeResponse initialize(const InitializeRequest& request) = 0;
    
    /// 创建新会话
    virtual SessionInfo new_session(const std::string& cwd, const std::vector<nlohmann::json>& mcp_servers = {}) = 0;
    
    /// 加载会话
    virtual std::optional<SessionInfo> load_session(const std::string& session_id) = 0;
    
    /// 处理提示
    virtual void prompt(
        const PromptRequest& request,
        std::function<void(const nlohmann::json& update)> on_update,
        std::function<void(StopReason, const Usage&)> on_complete
    ) = 0;
    
    /// 取消
    virtual void cancel(const std::string& session_id) = 0;
    
    /// 分叉会话
    virtual SessionInfo fork_session(const std::string& session_id) = 0;
    
    /// 设置模式
    virtual void set_mode(const std::string& session_id, const std::string& mode) = 0;
    
    /// 设置模型
    virtual void set_model(const std::string& session_id, const std::string& model_id) = 0;
};

/// ACP Client 能力
class TURBOT_CORE_API ACPClient {
public:
    virtual ~ACPClient() = default;
    
    /// 读取文件
    virtual std::future<std::string> read_text_file(const std::string& uri) = 0;
    
    /// 写入文件
    virtual std::future<void> write_text_file(const std::string& uri, const std::string& content) = 0;
    
    /// 请求权限
    virtual std::future<bool> request_permission(const PermissionOption& option) = 0;
    
    /// 显示终端输出
    virtual void show_terminal_output(const std::string& output) = 0;
};

/// ACP Server
class TURBOT_CORE_API ACPServer {
public:
    /// 启动 ACP 服务器 (stdio)
    static void start(std::unique_ptr<ACPAgent> agent, std::unique_ptr<ACPClient> client);
    
    /// 运行事件循环
    static void run();
    
    /// 停止服务器
    static void stop();
};

} // namespace turbot::core::acp
```
