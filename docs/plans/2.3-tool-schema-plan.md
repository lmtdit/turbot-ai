# 2.3 工具 Schema

> **依赖**: 无（独立）
> **工时**: 1 天
> **测试**: 单元测试

## 目标

将现有 Tool 定义转换为 LLM API 所需的 tools schema 格式。

## 输入

- 现有 `libs/core/tool/tool.hpp`（ToolRegistry）
- Agent 权限信息

## 参考

```typescript
// OpenCode tools schema 格式
const tools = registry.list().map((tool) => ({
  name: tool.name,
  description: tool.description,
  input_schema: tool.schema, // JSON Schema
}))
```

```cpp
// libs/core/session/tool_schema.hpp
#pragma once

#include <turbot/core/tool/tool.hpp>
#include <turbot/core/permission/permission.hpp>
#include <nlohmann/json.hpp>
#include <vector>
#include <string>

namespace turbot::core::session {

/// 工具定义（LLM API 格式）
struct ToolDefinition {
    std::string name;
    std::string description;
    nlohmann::json input_schema;  // JSON Schema

    nlohmann::json to_json() const;
};

/// 工具 Schema 构建器
class ToolSchemaBuilder {
public:
    /// 构建工具定义列表
    static std::vector<ToolDefinition> build(
        const tool::ToolRegistry& registry,
        const std::vector<std::string>& enabled_tools = {}
    );

    /// 过滤被禁用的工具
    static std::vector<std::string> filter_by_permission(
        const std::vector<std::string>& tools,
        const permission::Ruleset& ruleset
    );

    /// 转换为 OpenAI tools 格式
    static nlohmann::json to_openai_format(
        const std::vector<ToolDefinition>& tools
    );
};

} // namespace turbot::core::session
```

## 测试验收

```cpp
// libs/core/test/tool_schema_test.cpp

// 1. 单工具转换测试
TEST(ToolSchema, SingleTool) {
    auto& registry = tool::ToolRegistry::instance();
    auto tools = ToolSchemaBuilder::build(registry, {"read"});

    ASSERT_EQ(tools.size(), 1);
    EXPECT_EQ(tools[0].name, "read");
    EXPECT_FALSE(tools[0].description.empty());
    EXPECT_TRUE(tools[0].input_schema.contains("type"));
}

// 2. OpenAI 格式测试
TEST(ToolSchema, OpenAIFormat) {
    auto& registry = tool::ToolRegistry::instance();
    auto tools = ToolSchemaBuilder::build(registry, {"read", "write"});
    auto json = ToolSchemaBuilder::to_openai_format(tools);

    EXPECT_TRUE(json.is_array());
    EXPECT_EQ(json.size(), 2);
    EXPECT_TRUE(json[0].contains("type"));
    EXPECT_EQ(json[0]["type"], "function");
    EXPECT_TRUE(json[0].contains("function"));
}

// 3. 权限过滤测试
TEST(ToolSchema, PermissionFilter) {
    std::vector<std::string> tools = {"read", "write", "bash"};
    permission::Ruleset ruleset;
    // 设置 write 为 deny
    ruleset.set_rule("write", permission::PermissionAction::Deny);

    auto filtered = ToolSchemaBuilder::filter_by_permission(tools, ruleset);
    EXPECT_EQ(filtered.size(), 2);
    EXPECT_TRUE(std::find(filtered.begin(), filtered.end(), "read") != filtered.end());
    EXPECT_TRUE(std::find(filtered.begin(), filtered.end(), "bash") != filtered.end());
    EXPECT_FALSE(std::find(filtered.begin(), filtered.end(), "write") != filtered.end());
}

// 4. JSON Schema 验证测试
TEST(ToolSchema, SchemaValid) {
    auto& registry = tool::ToolRegistry::instance();
    auto tools = ToolSchemaBuilder::build(registry);

    for (const auto& tool : tools) {
        // 验证 schema 结构
        EXPECT_TRUE(tool.input_schema.contains("type"));
        EXPECT_EQ(tool.input_schema["type"], "object");
        // 验证必要字段
        EXPECT_TRUE(tool.input_schema.contains("properties"));
    }
}
```
