### 4.1 HTTP 客户端

> **实现说明**: 实际实现做了以下改进：
>
> - 使用 `HttpMethod` 枚举替代字符串，类型更安全

namespace turbot::network {

// HTTP 方法枚举（类型安全）
enum class HttpMethod {
    GET, POST, PUT, PATCH, DELETE_, HEAD, OPTIONS
};

// HTTP 请求
struct HttpRequest {
    HttpMethod method = HttpMethod::GET;
    std::string url;
    HttpHeaders headers;
    std::string body;
    int timeout_seconds = 30;
    bool follow_redirects = true;
    int max_redirects = 5;

    // 静态工厂方法
    static HttpRequest get(std::string_view url);
    static HttpRequest post(std::string_view url, std::string_view body = "");
    static HttpRequest put(std::string_view url, std::string_view body = "");
    static HttpRequest patch(std::string_view url, std::string_view body = "");
    static HttpRequest del(std::string_view url);

    // Builder 模式
    HttpRequest& with_header(std::string_view name, std::string_view value);
    HttpRequest& with_timeout(int seconds);
    HttpRequest& with_json_body(std::string_view json);
};

// HTTP 响应
struct HttpResponse {
    int status_code = 0;
    std::string status_message;
    std::string body;
    HttpHeaders headers;
    std::string content_type;
    long response_time_ms = 0;  // 响应时间（毫秒）

    // 状态检查方法
    bool is_success() const noexcept;      // 2xx
    bool is_client_error() const noexcept; // 4xx
    bool is_server_error() const noexcept; // 5xx

    // 获取响应头（大小写不敏感）
    std::optional<std::string> get_header(std::string_view name) const;
};

// 流式响应回调
tusing StreamCallback = std::function<bool(std::string_view chunk)>;

class HttpClient {
public:
    HttpClient();
    ~HttpClient();

    // 高级 API
    HttpResponse get(std::string_view url, const HttpHeaders& headers = {});
    HttpResponse post(std::string_view url, std::string_view body, const HttpHeaders& headers = {});
    HttpResponse post_json(std::string_view url, std::string_view json, const HttpHeaders& headers = {});
    HttpResponse put(std::string_view url, std::string_view body, const HttpHeaders& headers = {});
    HttpResponse patch(std::string_view url, std::string_view body, const HttpHeaders& headers = {});
    HttpResponse del(std::string_view url, const HttpHeaders& headers = {});

    // 低级 API
    HttpResponse request(const HttpRequest& req);
    HttpResponse request_stream(const HttpRequest& req, StreamCallback callback);

    // 异步 API（使用 std::future）
    std::future<HttpResponse> get_async(std::string_view url, const HttpHeaders& headers = {});
    std::future<HttpResponse> post_async(std::string_view url, std::string_view body, const HttpHeaders& headers = {});

    // 配置
    void set_timeout(int seconds);
    void set_default_header(std::string_view name, std::string_view value);
    void set_proxy(std::string_view proxy);
    void set_ssl_verify(bool verify);
    void set_user_agent(std::string_view user_agent);
};

} // namespace turbot::network

## 5. 实现要点

> **实现说明**: 实际实现采用同步阻塞模式（配合 std::async 实现异步），
> 流式响应使用回调模式，比原计划的协程方案更简单实用。

### 5.1 HTTP 客户端实现

```cpp
HttpResponse HttpClient::Impl::do_request(const HttpRequest& req, StreamCallback* stream_callback) {
    auto start_time = std::chrono::steady_clock::now();

    HttpResponse response;
    CurlData curl_data;
    curl_data.stream_callback = stream_callback;

    CURL* curl = curl_easy_init();
    if (!curl) {
        throw std::runtime_error("Failed to initialize CURL handle");
    }

    // 设置 URL
    curl_easy_setopt(curl, CURLOPT_URL, req.url.c_str());

    // 设置方法（使用枚举转字符串）
    curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, method_to_string(req.method));

    // 设置超时
    int timeout = req.timeout_seconds > 0 ? req.timeout_seconds : default_timeout_;
    curl_easy_setopt(curl, CURLOPT_TIMEOUT, timeout);
    curl_easy_setopt(curl, CURLOPT_CONNECTTIMEOUT, timeout);

    // 设置重定向
    if (req.follow_redirects) {
        curl_easy_setopt(curl, CURLOPT_FOLLOWLOCATION, 1L);
        curl_easy_setopt(curl, CURLOPT_MAXREDIRS, req.max_redirects);
    }

    // 设置请求头（合并默认头）
    struct curl_slist* curl_headers = nullptr;
    for (const auto& [key, value] : default_headers_) {
        curl_headers = curl_slist_append(curl_headers, (key + ": " + value).c_str());
    }
    for (const auto& [key, value] : req.headers) {
        curl_headers = curl_slist_append(curl_headers, (key + ": " + value).c_str());
    }
    curl_easy_setopt(curl, CURLOPT_HTTPHEADER, curl_headers);

    // 设置请求体
    if (!req.body.empty()) {
        curl_easy_setopt(curl, CURLOPT_POSTFIELDS, req.body.c_str());
        curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE, req.body.size());
    }

    // 设置回调
    curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, write_body_callback);
    curl_easy_setopt(curl, CURLOPT_WRITEDATA, &curl_data);
    curl_easy_setopt(curl, CURLOPT_HEADERFUNCTION, write_header_callback);
    curl_easy_setopt(curl, CURLOPT_HEADERDATA, &curl_data);

    // 执行请求
    CURLcode res = curl_easy_perform(curl);

    // 处理结果
    if (res != CURLE_OK || curl_data.aborted) {
        curl_slist_free_all(curl_headers);
        curl_easy_cleanup(curl);
        throw std::runtime_error(curl_easy_strerror(res));
    }

    // 获取响应码
    long status_code;
    curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, &status_code);
    response.status_code = static_cast<int>(status_code);

    // 解析响应头
    response.headers = parse_headers(curl_data.headers);
    response.body = curl_data.body;

    // 计算响应时间
    auto end_time = std::chrono::steady_clock::now();
    response.response_time_ms = std::chrono::duration_cast<std::chrono::milliseconds>(end_time - start_time).count();

    curl_slist_free_all(curl_headers);
    curl_easy_cleanup(curl);

    return response;
}
```

### 5.2 流式响应处理（回调模式）

```cpp
// 流式回调 - 每个 chunk 调用一次，返回 false 中止传输
static size_t write_body_callback(void* contents, size_t size, size_t nmemb, void* userp) {
    size_t total_size = size * nmemb;
    auto* data = static_cast<CurlData*>(userp);

    if (data->stream_callback) {
        // 流式模式：传递给回调
        if (!(*data->stream_callback)(std::string_view(static_cast<const char*>(contents), total_size))) {
            data->aborted = true;
            return 0;  // 中止传输
        }
    } else {
        // 缓冲模式：追加到 body
        data->body.append(static_cast<const char*>(contents), total_size);
    }
    return total_size;
}

// 使用示例：处理 SSE 流
HttpClient client;
std::string collected;
client.request_stream(req, [&](std::string_view chunk) {
    collected.append(chunk);
    // 处理 SSE 数据
    if (chunk.find("data: [DONE]") != std::string_view::npos) {
        return false;  // 完成，中止
    }
    return true;  // 继续
});
```

HttpClient client;
