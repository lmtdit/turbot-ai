## 2. Agent 系统

### 2.1 Agent 接口

```cpp
namespace turbot::core::agent {

enum class AgentMode {
    Primary,    // 主要代理
    Subagent,   // 子代理
    All
};

struct AgentInfo {
    std::string name;
    std::optional<std::string> description;
    AgentMode mode;
    bool native = false;
    bool hidden = false;
    nlohmann::json permission;
    std::optional<std::string> model_id;
    nlohmann::json options;
};

class Agent {
public:
    virtual ~Agent() = default;

    static void register_agent(std::unique_ptr<Agent> agent);

    static Task<std::optional<Agent>> get(const std::string& name);

    static Task<std::vector<Agent>> list();

    virtual std::string name() const = 0;

    virtual Task<nlohmann::json> execute(const ExecuteParams& params) = 0;

    const AgentInfo& info() const { return info_; }

protected:
    AgentInfo info_;
};

class AgentRegistry {
public:
    static AgentRegistry& instance();

    void register_agent(std::unique_ptr<Agent> agent);

    std::shared_ptr<Agent> get(const std::string& name);

    std::vector<std::shared_ptr<Agent>> list();

private:
    std::unordered_map<std::string, std::shared_ptr<Agent>> agents_;
};

} // namespace turbot::core::agent
```

### 2.2 内置代理

```cpp
// Build Agent
class BuildAgent : public Agent {
public:
    BuildAgent() {
        info_.name = "build";
        info_.description = "Default agent for executing tools";
        info_.mode = AgentMode::Primary;
        info_.permission = build_default_permission();
    }

    std::string name() const override { return info_.name; }

    Task<nlohmann::json> execute(const ExecuteParams& params) override;

private:
    nlohmann::json build_default_permission() {
        return {
            {{"permission", "*"}, {"pattern", "*"}, {"action", "allow"}},
            {{"permission", "doom_loop"}, {"pattern", "*"}, {"action", "ask"}}
        };
    }
};

// Plan Agent
class PlanAgent : public Agent {
public:
    PlanAgent() {
        info_.name = "plan";
        info_.description = "Agent for planning and analysis";
        info_.mode = AgentMode::Primary;
    }

    std::string name() const override { return info_.name; }

    Task<nlohmann::json> execute(const ExecuteParams& params) override;
};

// Explore Agent
class ExploreAgent : public Agent {
public:
    ExploreAgent() {
        info_.name = "explore";

        info_.mode = AgentMode::Primary;

    std::string name() const override { return info_.name; }

    Task<nlohmann::json> execute(const ExecuteParams& params) override;
```

## 3. Session 系统

### 3.1 Session 接口

```cpp
namespace turbot::core::session {

enum class SessionState {
    Created,
    Active,
    Busy,
    Compacting,
    Archived
};

struct SessionInfo {
    std::string id;
    std::string project_id;
    std::optional<std::string> parent_id;
    std::string slug;
    std::string directory;
    std::string title;
    std::string version;
    std::optional<nlohmann::json> permission;
    int64_t time_created;
    int64_t time_updated;
    std::optional<int64_t> time_compacting;
    std::optional<int64_t> time_archived;
};

class Session {
public:
    static Task<Session> create(const CreateParams& params);

    static Task<Session> fork(const ForkParams& params);

    static Task<std::optional<Session>> get(const std::string& id);

    static Task<std::vector<Session>> list(const std::string& project_id);

    Task<void> update(const UpdateParams& params);

    Task<void> set_title(const std::string& title);

    Task<void> set_permission(const nlohmann::json& permission);

    Task<std::vector<message::Message>> messages(int limit = 50, int offset = 0);

    Task<void> compact();

    Task<void> archive();

    const std::string& id() const { return info_.id; }

    const SessionInfo& info() const { return info_; }

private:
    SessionInfo info_;
    std::shared_ptr<storage::Database> db_;
};

} // namespace turbot::core::session
```

## 4. 使用示例

```cpp
// 注册内置代理
AgentRegistry::instance().register_agent(std::make_unique<BuildAgent>());
AgentRegistry::instance().register_agent(std::make_unique<PlanAgent>());
AgentRegistry::instance().register_agent(std::make_unique<ExploreAgent>());

// 创建会话
auto session = co_await Session::create({
    .project_id = "project123",
    .slug = "main",
    .directory = "/path/to/project",
    .title = "Main Session"
});

// 获取代理
auto agent = co_await Agent::get("build");

// 执行代理任务
auto result = co_await agent->execute({
    .session_id = session.id(),
    .prompt = "Write a function to add two numbers"
});
```
