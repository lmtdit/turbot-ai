# Turbot-AI 多Agent协作架构技术实现方案

## 1. 项目概述

### 1.1 背景

基于对 OpenCode 项目架构的深入分析，Turbot-AI 项目旨在使用 C++23 实现一个类似的多 Agent 协作 AI 开发助手框架。本项目将借鉴 OpenCode 的核心设计理念，结合 C++ 的高性能特性，构建一个高效、安全、可扩展的 AI 编码助手系统。

### 1.2 设计目标

- **高性能**: 利用 C++ 的性能优势，提供低延迟的 AI 交互体验
- **模块化**: 清晰的模块边界，易于维护和扩展
- **类型安全**: 利用 C++23 的强类型系统和概念约束
- **跨平台**: 支持主流操作系统（Linux, macOS, Windows）
- **安全性**: 完善的权限系统和安全机制
- **可观测性**: 内置日志、监控和追踪能力

### 1.3 核心特性

- 多 Agent 协作机制
- 流式 AI 响应处理
- 智能会话压缩
- 灵活的工具系统
- 强大的权限控制
- 事件驱动架构
- 持久化存储

```cpp
// libs/core/event/bus.hpp
namespace turbot::core::event {

template<typename T>
struct Event {
    std::string name;
    T data;
    int64_t timestamp;
};

template<typename T>
using EventHandler = std::function<Task<void>(const Event<T>&)>;

class EventBus {
public:
    // 单例访问
    static EventBus& instance();

    // 发布事件
    template<typename T>
    Task<void> publish(const std::string& name, T data) {
        Event<T> event{
            .name = name,
            .data = std::move(data),
            .timestamp = std::chrono::system_clock::now().time_since_epoch().count()
        };

        std::shared_lock lock(mutex_);
        auto it = handlers_.find(name);
        if (it != handlers_.end()) {
            for (auto& handler : it->second) {
                co_await handler(event);
            }
        }
    }

    // 订阅事件
    template<typename T>
    std::string subscribe(const std::string& name, EventHandler<T> handler) {
        std::string id = generate_id();
        std::unique_lock lock(mutex_);
        handlers_[name].push_back({
            .id = id,
            .handler = [handler](const std::any& event) -> Task<void> {
                co_await handler(std::any_cast<const Event<T>&>(event));
            }
        });
        return id;
    }

    // 取消订阅
    void unsubscribe(const std::string& name, const std::string& handler_id);

private:
    struct HandlerEntry {
        std::string id;
        std::function<Task<void>(const std::any&)> handler;
    };

    EventBus() = default;
    std::unordered_map<std::string, std::vector<HandlerEntry>> handlers_;
    mutable std::shared_mutex mutex_;
};

} // namespace turbot::core::event
```

### 4.1 Task工具实现

```cpp
// tools/task/task_tool.hpp
namespace turbot::tools::task {

struct TaskParams {
    std::string prompt;
    std::string description;
    std::string subagent_type;
    std::optional<std::string> task_id;
    std::optional<std::string> command;
};

class TaskTool : public core::tool::Tool {
public:
    std::string name() const override { return "task"; }

    std::string description() const override {
        return "Execute a subtask using a subagent";
    }

    nlohmann::json input_schema() const override {
        return R"({
            "type": "object",
            "properties": {
                "prompt": {"type": "string"},
                "description": {"type": "string"},
                "subagent_type": {"type": "string"},
                "task_id": {"type": "string"},
                "command": {"type": "string"}
            },
            "required": ["prompt", "description", "subagent_type"]
        })"_json;
    }

    Task<ToolResult> execute(const nlohmann::json& input,
                             ToolContext& ctx) override {
        TaskParams params = input.get<TaskParams>();

        // 权限检查
        if (!ctx.extra.contains("bypassAgentCheck")) {
            co_await ctx.ask({
                .permission = "task",
                .patterns = {params.subagent_type},
                .always = {"*"},
                .metadata = {
                    {"description", params.description},
                    {"subagent_type", params.subagent_type}
                }
            });
        }

        // 获取子代理
        auto agent = co_await agent::Agent::get(params.subagent_type);
        if (!agent) {
            throw std::runtime_error("Agent not found: " + params.subagent_type);
        }

        // 创建或恢复子会话
        session::Session sub_session;
        if (params.task_id) {
            auto existing = co_await session::Session::get(*params.task_id);
            if (existing) {
                sub_session = std::move(*existing);
            }
        }

        if (!sub_session) {
            // 创建新子会话
            sub_session = co_await session::Session::create({
                .parent_id = ctx.session_id,
                .title = params.description + " (@" + agent->name() + " subagent)",
                .permission = build_subagent_permission(*agent)
            });
        }

        // 在子会话中执行任务
        auto result = co_await execute_in_subsession(sub_session, *agent, params, ctx);

        co_return ToolResult{
            .title = params.description,
            .output = format_task_result(result, sub_session.id()),
            .metadata = {
                {"session_id", sub_session.id()},
                {"subagent", params.subagent_type}
            }
        };
    }

private:
    permission::Ruleset build_subagent_permission(const agent::Agent& agent) {
        // 限制子会话权限
        return {
            {{"permission", "todowrite"}, {"pattern", "*"}, {"action", "deny"}},
            {{"permission", "todoread"}, {"pattern", "*"}, {"action", "deny"}},
            {{"permission", "task"}, {"pattern", "*"}, {"action", "deny"}}
        };
    }

    Task<std::string> execute_in_subsession(
        session::Session& session,
        const agent::Agent& agent,
        const TaskParams& params,
        ToolContext& ctx) {
        // 创建用户消息
        auto user_msg = co_await message::Message::create({
            .session_id = session.id(),
            .role = message::Role::User,
            .content = params.prompt
        });

        // 执行会话主循环
        auto assistant_msg = co_await session::Session::prompt({
            .session_id = session.id(),
            .message_id = user_msg.id(),
            .agent = agent.name()
        });

        co_return assistant_msg.to_text();
    }

    std::string format_task_result(const std::string& result, const std::string& session_id) {
        return "task_id: " + session_id + "\n<task_result>\n" + result + "\n</task_result>";
    }
};

} // namespace turbot::tools::task
```

```cpp
class SessionLoop {
public:
    SessionLoop(const std::string& session_id) : session_id_(session_id) {}

    Task<void> run() {
        int step = 0;
        while (true) {
            // 获取消息历史
            auto messages = co_await Message::stream(session_id_);

            // 找到最后一条用户消息和助手消息
            auto [last_user, last_assistant] = find_last_messages(messages);

            // 收集待处理任务
            auto pending_tasks = collect_pending_tasks(messages);

            // 处理路径选择
            if (!pending_tasks.empty()) {
                if (auto* task = std::get_if<SubtaskPart>(&pending_tasks.back())) {
                    co_await handle_subtask(*task, last_user);
                    continue;
                }
                if (auto* task = std::get_if<CompactionPart>(&pending_tasks.back())) {
                    co_await handle_compaction(*task);
                    continue;
                }
            }

            // 检查是否需要压缩
            if (should_compact(messages)) {
                co_await create_compaction_task();
                continue;
            }

            // 正常LLM处理
            auto result = co_await process_llm_request(messages);
            if (result == LoopResult::Stop) {
                break;
            }

            step++;
        }
    }

private:
    std::string session_id_;

    enum class LoopResult {
        Continue,
        Stop,
        Compact
    };

    Task<void> handle_subtask(const SubtaskPart& task, const Message* last_user) {
        // 创建助手消息
        auto assistant_msg = co_await Message::create({
            .session_id = session_id_,
            .role = Role::Assistant,
            .parent_id = last_user->id(),
            .agent = task.agent
        });

        // 执行Task工具
        TaskTool task_tool;
        auto result = co_await task_tool.execute(task.params, {
            .session_id = session_id_,
            .message_id = assistant_msg.id(),
            .agent = task.agent
        });

        // 更新消息
        assistant_msg.add_part({
            .type = PartType::Tool,
            .tool = "task",
            .state = ToolState::Completed,
            .output = result.output
        });

        co_await assistant_msg.save();
    }

    Task<LoopResult> process_llm_request(const std::vector<Message>& messages) {
        // 获取代理
        auto agent = co_await Agent::get(get_agent_name(messages));

        // 获取模型
        auto model = co_await Provider::get_default_model();

        // 获取可用工具
        auto tools = co_await ToolRegistry::instance().filter(*agent);

        // 构建系统提示词
        auto system = build_system_prompt(*agent);

        // 调用LLM
        auto stream = co_await model->chat_stream(
            to_model_messages(messages, system),
            {model->id(), model->provider_id()}
        );

        // 处理流式响应
        auto assistant_msg = co_await Message::create({
            .session_id = session_id_,
            .role = Role::Assistant
        });

        bool has_tool_calls = false;
        for (auto event : stream) {
            switch (event.type) {
                case ChatStreamEvent::Type::TextDelta:
                    assistant_msg.append_text(event.content);
                    break;
                case ChatStreamEvent::Type::ToolCall:
                    has_tool_calls = true;
                    co_await handle_tool_call(event.data, assistant_msg);
                    break;
                case ChatStreamEvent::Type::Finish:
                    assistant_msg.set_finish(event.content);
                    break;
                case ChatStreamEvent::Type::Error:
                    assistant_msg.set_error(event.data);
                    break;
            }
        }

        co_await assistant_msg.save();

        return has_tool_calls ? LoopResult::Continue : LoopResult::Stop;
    }
};

} // namespace turbot::core::session
```

```cpp
// libs/storage/sqlite/sqlite_database.cpp
namespace turbot::storage::sqlite {

class SQLiteDatabase : public Database {
public:
    SQLiteDatabase(const std::string& path) {
        int rc = sqlite3_open(path.c_str(), &db_);
        if (rc != SQLITE_OK) {
            throw std::runtime_error("Cannot open database: " + std::string(sqlite3_errmsg(db_)));
        }
    }

    ~SQLiteDatabase() override {
        if (db_) {
            sqlite3_close(db_);
        }
    }

    Task<QueryResult> execute(const std::string& sql,
                             const std::vector<nlohmann::json>& params) override {
        sqlite3_stmt* stmt = nullptr;
        int rc = sqlite3_prepare_v2(db_, sql.c_str(), -1, &stmt, nullptr);
        if (rc != SQLITE_OK) {
            throw std::runtime_error("Prepare failed: " + std::string(sqlite3_errmsg(db_)));
        }

        // 绑定参数
        for (size_t i = 0; i < params.size(); i++) {
            bind_param(stmt, i + 1, params[i]);
        }

        // 执行查询
        QueryResult result;
        while ((rc = sqlite3_step(stmt)) == SQLITE_ROW) {
            result.rows.push_back(row_to_json(stmt));
        }

        result.affected_rows = sqlite3_changes(db_);
        sqlite3_finalize(stmt);

        if (rc != SQLITE_DONE) {
            throw std::runtime_error("Step failed: " + std::string(sqlite3_errmsg(db_)));
        }

        co_return result;
    }

private:
    sqlite3* db_ = nullptr;

    void bind_param(sqlite3_stmt* stmt, int index, const nlohmann::json& value) {
        if (value.is_null()) {
            sqlite3_bind_null(stmt, index);
        } else if (value.is_string()) {
            sqlite3_bind_text(stmt, index, value.get<std::string>().c_str(), -1, SQLITE_TRANSIENT);
        } else if (value.is_number_integer()) {
            sqlite3_bind_int64(stmt, index, value.get<int64_t>());
        } else if (value.is_number_float()) {
            sqlite3_bind_double(stmt, index, value.get<double>());
        } else if (value.is_boolean()) {
            sqlite3_bind_int(stmt, index, value.get<bool>() ? 1 : 0);
        } else {
            sqlite3_bind_text(stmt, index, value.dump().c_str(), -1, SQLITE_TRANSIENT);
        }
    }

    nlohmann::json row_to_json(sqlite3_stmt* stmt) {
        nlohmann::json row;
        int count = sqlite3_column_count(stmt);
        for (int i = 0; i < count; i++) {
            const char* name = sqlite3_column_name(stmt, i);
            switch (sqlite3_column_type(stmt, i)) {
                case SQLITE_TEXT:
                    row[name] = reinterpret_cast<const char*>(sqlite3_column_text(stmt, i));
                    break;
                case SQLITE_INTEGER:
                    row[name] = sqlite3_column_int64(stmt, i);
                    break;
                case SQLITE_FLOAT:
                    row[name] = sqlite3_column_double(stmt, i);
                    break;
                case SQLITE_NULL:
                    row[name] = nullptr;
                    break;
            }
        }
        return row;
    }
};

} // namespace turbot::storage::sqlite
```

```cpp
// libs/storage/migration/migration.hpp
namespace turbot::storage::migration {

class Migration {
public:
    virtual std::string name() const = 0;
    virtual std::string up() const = 0;
    virtual std::string down() const = 0;
};

class MigrationRunner {
public:
    MigrationRunner(std::shared_ptr<Database> db) : db_(std::move(db)) {}

    Task<void> run() {
        // 创建迁移表
        co_await create_migration_table();

        // 获取已执行的迁移
        auto executed = co_await get_executed_migrations();

        // 执行新迁移
        for (const auto& migration : migrations_) {
            if (executed.contains(migration->name())) {
                continue;
            }

            TURBOT_LOG_INFO("Running migration: {}", migration->name());

            co_await db_->execute(migration->up());
            co_await record_migration(migration->name());
        }
    }

    void add_migration(std::unique_ptr<Migration> migration) {
        migrations_.push_back(std::move(migration));
    }

private:
    std::shared_ptr<Database> db_;
    std::vector<std::unique_ptr<Migration>> migrations_;

    Task<void> create_migration_table() {
        co_await db_->execute(R"(
            CREATE TABLE IF NOT EXISTS schema_migrations (
                name TEXT PRIMARY KEY,
                executed_at INTEGER NOT NULL
            )
        )");
    }

    Task<std::unordered_set<std::string>> get_executed_migrations() {
        auto result = co_await db_->execute("SELECT name FROM schema_migrations");
        std::unordered_set<std::string> migrations;
        for (const auto& row : result.rows) {
            migrations.insert(row["name"].get<std::string>());
        }
        co_return migrations;
    }

    Task<void> record_migration(const std::string& name) {
        co_await db_->execute(
            "INSERT INTO schema_migrations (name, executed_at) VALUES (?, ?)",
            {name, std::chrono::system_clock::now().time_since_epoch().count()}
        );
    }
};

} // namespace turbot::storage::migration
```
